HOW IT WORKS
- Each bubble owns a private VirtualDisplay (public DisplayManager API). Its pixels are drawn into a SurfaceView inside our own overlay window, so we own the frame, minimize, close, move and resize (VirtualDisplay.resize).
- Shizuku (shell UID) starts the app on that display with "am start --display <id>" and injects ALL touch pointers with the hidden input service, so pinch, double tap and flings are handled by the app itself.
- Arrange mode tiles open windows in bubble order: 1 fills the area, 2 stack (portrait) or sit side by side (landscape), 3 is two on top and one full-width below.
- The chip is an app drawer built as our own overlay (a service cannot show a bottom sheet). It becomes focusable only while open so the search field can use the keyboard.
- Fallback: if the ROM lacks the secondary-display feature, or launching on our display fails, bubbles switch to the system freeform window. Arrange then resizes those tasks with am task resize, best effort.
- BubbleManager is pure state (cap of 3, recents, order, persistence) and unit tested. Pass singleOpen = true to its constructor for strict Android 17 behavior (one window at a time).

KNOWN GAPS
- Keyboard typing inside embedded apps is not supported yet. Taps, scrolling, pinch and the back button work.
- Several embedded apps at once cost RAM and GPU. If the phone lags, lower MAX_BUBBLES in BubbleManager.kt or use singleOpen.
- No manual window resize handle yet (arrange mode resizes).
- Dragging a window does not persist its free position.

package com.multitask.bubble.core

import android.content.Context

/**
 * Pure bubble state: the list, the cap, recents and persistence. Main thread only, never blocks.
 * The caller reacts to the returned [Change] (hide or release windows).
 *
 * With [singleOpen] true opening a bubble collapses the others, like Android 17. The default lets
 * several windows stay open so they can be tiled.
 */
class BubbleManager(
    context: Context,
    private val singleOpen: Boolean = false,
    private val onChanged: () -> Unit
) {
    data class Bubble(val pkg: String, val expanded: Boolean = false)

    /** [evicted] bubbles were removed to make room. [collapsed] bubbles were open and got hidden. */
    data class Change(
        val evicted: List<String> = emptyList(),
        val collapsed: List<String> = emptyList()
    )

    companion object {
        const val MAX_BUBBLES = 3
        const val MAX_RECENTS = 8
        private const val PREFS = "bubbles"
        private const val KEY_ACTIVE = "active"
        private const val KEY_RECENTS = "recents"
    }

    private val prefs = context.applicationContext.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
    private val bubbles = mutableListOf<Bubble>()
    private val recents = mutableListOf<String>()

    init {
        restore()
    }

    fun snapshot(): List<Bubble> = bubbles.toList()

    fun recentsSnapshot(): List<String> = recents.toList()

    /** Adds the app if needed (evicting the oldest at the cap) and makes it the one open bubble. */
    fun open(pkg: String): Change {
        val evicted = mutableListOf<String>()
        if (bubbles.none { it.pkg == pkg }) {
            if (bubbles.size >= MAX_BUBBLES) {
                val oldest = bubbles.first().pkg
                remove(oldest)
                evicted.add(oldest)
            }
            bubbles.add(Bubble(pkg))
            recents.remove(pkg)
        }
        val collapsed = if (singleOpen) {
            bubbles.filter { it.expanded && it.pkg != pkg }.map { it.pkg }
        } else {
            emptyList()
        }
        bubbles.replaceAll {
            when {
                it.pkg == pkg -> it.copy(expanded = true)
                singleOpen -> it.copy(expanded = false)
                else -> it
            }
        }
        commit()
        return Change(evicted, collapsed)
    }

    fun collapse(pkg: String) {
        bubbles.replaceAll { if (it.pkg == pkg) it.copy(expanded = false) else it }
        commit()
    }

    /** Moves the bubble so it ends up at [index] in the list. Order decides the tiling order. */
    fun moveTo(pkg: String, index: Int) {
        val from = bubbles.indexOfFirst { it.pkg == pkg }
        if (from < 0) return
        val item = bubbles.removeAt(from)
        bubbles.add(index.coerceIn(0, bubbles.size), item)
        commit()
    }

    /** Removes the bubble and remembers it in recents. */
    fun dismiss(pkg: String) {
        if (bubbles.none { it.pkg == pkg }) return
        remove(pkg)
        commit()
    }

    private fun remove(pkg: String) {
        bubbles.removeAll { it.pkg == pkg }
        recents.remove(pkg)
        recents.add(0, pkg)
        while (recents.size > MAX_RECENTS) recents.removeAt(recents.lastIndex)
    }

    private fun commit() {
        prefs.edit()
            .putString(KEY_ACTIVE, bubbles.joinToString(",") { it.pkg })
            .putString(KEY_RECENTS, recents.joinToString(","))
            .apply()
        onChanged()
    }

    /** Restored bubbles start collapsed; their apps start fresh on first tap. */
    private fun restore() {
        prefs.getString(KEY_ACTIVE, "").orEmpty().split(",").filter { it.isNotBlank() }
            .take(MAX_BUBBLES).forEach { bubbles.add(Bubble(it)) }
        prefs.getString(KEY_RECENTS, "").orEmpty().split(",").filter { it.isNotBlank() }
            .take(MAX_RECENTS).forEach { recents.add(it) }
    }
}

package com.multitask.bubble.core

import android.util.Log
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

/** In-memory log shared by the activity and the overlay service, mirrored to logcat. */
object DiagLog {
    private const val MAX_CHARS = 6000
    private const val TRIM_CHARS = 2000

    private val buffer = StringBuilder()

    /** Called on whichever thread logged. UI code must post to the main thread itself. */
    @Volatile
    var listener: ((String) -> Unit)? = null

    fun log(tag: String, message: String) {
        Log.d(tag, message)
        val stamp = SimpleDateFormat("HH:mm:ss", Locale.getDefault()).format(Date())
        synchronized(buffer) {
            buffer.append('[').append(stamp).append("] ").append(message).append('\n')
            if (buffer.length > MAX_CHARS) buffer.delete(0, TRIM_CHARS)
        }
        listener?.invoke(snapshot())
    }

    fun snapshot(): String = synchronized(buffer) { buffer.toString() }

    fun clear() {
        synchronized(buffer) { buffer.setLength(0) }
        listener?.invoke("")
    }
}

 * Runs inside the Shizuku shell process (UID 2000). Hidden APIs are reachable here because this
 * is not a normal app process.
 */
class ShellService : IShellService.Stub() {

    private val inputManager: Any? by lazy {
        runCatching {
            val binder = Class.forName("android.os.ServiceManager")
                .getMethod("getService", String::class.java)
                .invoke(null, "input") as IBinder
            Class.forName("android.hardware.input.IInputManager\$Stub")
                .getMethod("asInterface", IBinder::class.java)
                .invoke(null, binder)
        }.getOrNull()
    }

    private val injectMethod by lazy {
        runCatching {
            Class.forName("android.hardware.input.IInputManager")
                .getMethod("injectInputEvent", InputEvent::class.java, Int::class.javaPrimitiveType)
        }.getOrNull()
    }

    private val setDisplayId by lazy {
        runCatching {
            InputEvent::class.java.getMethod("setDisplayId", Int::class.javaPrimitiveType)
        }.getOrNull()
    }

    override fun destroy() {
        exitProcess(0)
    }

    override fun exec(command: String): String {
        return try {
            val process = ProcessBuilder("sh", "-c", command)
                .redirectErrorStream(true)
                .start()
            val output = process.inputStream.bufferedReader().readText()
            process.waitFor()
            output
        } catch (e: Exception) {
            "[exception] ${e.message}"
        }
    }

    /**
     * Forwards a whole touch frame. [action] is the raw MotionEvent action, so it already carries
     * the pointer index for POINTER_DOWN/UP. The target app does its own pinch and double-tap
     * detection from these raw pointers.
     */
    override fun injectTouch(
        displayId: Int, action: Int, downTime: Long,
        ids: IntArray, xs: FloatArray, ys: FloatArray
    ): Boolean {
        val count = ids.size
        if (count == 0 || xs.size != count || ys.size != count) return false
        val properties = Array(count) {
            MotionEvent.PointerProperties().apply {
                id = ids[it]
                toolType = MotionEvent.TOOL_TYPE_FINGER
            }
        }
        val coords = Array(count) {
            MotionEvent.PointerCoords().apply {
                x = xs[it]
                y = ys[it]
                pressure = 1f
                size = 1f
            }
        }
        val event = MotionEvent.obtain(
            downTime, SystemClock.uptimeMillis(), action, count, properties, coords,
            0, 0, 1f, 1f, 0, 0, InputDevice.SOURCE_TOUCHSCREEN, 0
        )
        return try {
            inject(event, displayId)
        } finally {
            event.recycle()
        }
    }

    override fun injectKey(displayId: Int, keyCode: Int): Boolean {
        val now = SystemClock.uptimeMillis()
        val down = KeyEvent(now, now, KeyEvent.ACTION_DOWN, keyCode, 0)
        val up = KeyEvent(now, now, KeyEvent.ACTION_UP, keyCode, 0)
        down.source = InputDevice.SOURCE_KEYBOARD
        up.source = InputDevice.SOURCE_KEYBOARD
        return inject(down, displayId) && inject(up, displayId)
    }

    private fun inject(event: InputEvent, displayId: Int): Boolean = try {
        setDisplayId?.invoke(event, displayId)
        // 0 = INJECT_INPUT_EVENT_MODE_ASYNC
        injectMethod?.invoke(inputManager, event, 0) as? Boolean ?: false
    } catch (e: Exception) {
        false
    }
}

package com.multitask.bubble.core

import android.content.ComponentName
import android.content.Context
import android.content.ServiceConnection
import android.content.pm.PackageManager
import android.graphics.Rect
import android.os.IBinder
import rikka.shizuku.Shizuku

class ShizukuWindowBackend(private val context: Context) : WindowBackend {

    companion object {
        const val TAG = "ShizukuBackend"
        private const val REQUEST_CODE = 1001
        private const val LOG_OUTPUT_LIMIT = 300
        private const val WINDOWING_MODE_FREEFORM = 5
        private const val TASK_POLL_ATTEMPTS = 8
        private const val TASK_POLL_DELAY_MS = 250L
    }

    @Volatile
    private var shell: IShellService? = null

    @Volatile
    private var prepared = false

    @Volatile
    private var injectFailureLogged = false

    private val waiting = mutableListOf<() -> Unit>()

    // Bump version() whenever ShellService or its AIDL changes, or Shizuku keeps the old process.
    private val userServiceArgs = Shizuku.UserServiceArgs(
        ComponentName(context.packageName, ShellService::class.java.name)
    ).daemon(false).processNameSuffix("shell").debuggable(false).version(4)

    private val connection = object : ServiceConnection {
        override fun onServiceConnected(name: ComponentName?, binder: IBinder?) {
            shell = IShellService.Stub.asInterface(binder)
            DiagLog.log(TAG, "Shell service connected")
            val callbacks = synchronized(waiting) { waiting.toList().also { waiting.clear() } }
            callbacks.forEach { it() }
        }

        override fun onServiceDisconnected(name: ComponentName?) {
            shell = null
            prepared = false
            DiagLog.log(TAG, "Shell service disconnected")
        }
    }

    override fun isReady(): Boolean = try {
        Shizuku.pingBinder() && Shizuku.checkSelfPermission() == PackageManager.PERMISSION_GRANTED
    } catch (e: Exception) {
        false
    }

    override fun requestPermission(onResult: (Boolean) -> Unit) {
        if (!Shizuku.pingBinder()) {
            onResult(false)
            return
        }
        if (Shizuku.checkSelfPermission() == PackageManager.PERMISSION_GRANTED) {
            onResult(true)
            return
        }
        val listener = object : Shizuku.OnRequestPermissionResultListener {
            override fun onRequestPermissionResult(requestCode: Int, grantResult: Int) {
                if (requestCode != REQUEST_CODE) return
                Shizuku.removeRequestPermissionResultListener(this)
                val granted = grantResult == PackageManager.PERMISSION_GRANTED
                DiagLog.log(TAG, "Shizuku permission ${if (granted) "granted" else "denied"}")
                onResult(granted)
            }
        }
        Shizuku.addRequestPermissionResultListener(listener)
        Shizuku.requestPermission(REQUEST_CODE)
    }

    override fun connect(onConnected: () -> Unit) {
        if (shell != null) {
            onConnected()
            return
        }
        if (!isReady()) {
            DiagLog.log(TAG, "Cannot connect: Shizuku is not running or not authorized")
            return
        }
        val firstWaiter = synchronized(waiting) {
            waiting.add(onConnected)
            waiting.size == 1
        }
        if (firstWaiter) {
            DiagLog.log(TAG, "Binding shell service")
            Shizuku.bindUserService(userServiceArgs, connection)
        }
    }

    override fun disconnect() {
        if (shell == null) return
        runCatching { Shizuku.unbindUserService(userServiceArgs, connection, true) }
        shell = null
        prepared = false
    }

    override fun prepare() {
        if (prepared) return
        // Lets apps that are not marked resizable fill a secondary display or a freeform window.
        sh("settings put global force_resizable_activities 1")
        sh("settings put global enable_freeform_support 1")
        prepared = true
    }

    override fun launchOnDisplay(packageName: String, displayId: Int): Boolean {
        val component = context.packageManager.getLaunchIntentForPackage(packageName)
            ?.component?.flattenToShortString()
        if (component == null) {
            DiagLog.log(TAG, "No launcher activity found for $packageName")
            return false
        }
        val output = sh("am start -n $component --display $displayId")
        val failed = output.contains("Error", ignoreCase = true) ||
            output.contains("Exception", ignoreCase = true)
        if (failed) DiagLog.log(TAG, "Launch on display $displayId failed for $packageName")
        return !failed
    }

    override fun launchFreeform(packageName: String, bounds: Rect): Int {
        val component = context.packageManager.getLaunchIntentForPackage(packageName)
            ?.component?.flattenToShortString()
        if (component == null) {
            DiagLog.log(TAG, "No launcher activity found for $packageName")
            return -1
        }
        sh("am start -n $component --windowingMode $WINDOWING_MODE_FREEFORM")
        var taskId = -1
        for (attempt in 0 until TASK_POLL_ATTEMPTS) {
            taskId = findTaskId(packageName)
            if (taskId > 0) break
            Thread.sleep(TASK_POLL_DELAY_MS)
        }
        if (taskId > 0) moveTask(taskId, bounds)
        else DiagLog.log(TAG, "Could not find a task for $packageName, so it cannot be moved")
        return taskId
    }

    override fun moveTask(taskId: Int, bounds: Rect) {
        sh("am task resize $taskId ${bounds.left},${bounds.top},${bounds.right},${bounds.bottom}")
    }

    override fun injectTouch(
        displayId: Int, action: Int, downTime: Long,
        ids: IntArray, xs: FloatArray, ys: FloatArray
    ) {
        val ok = try {
            shell?.injectTouch(displayId, action, downTime, ids, xs, ys) ?: false
        } catch (e: Exception) {
            false
        }
        if (!ok) logInjectFailureOnce()
    }

    override fun injectKey(displayId: Int, keyCode: Int) {
        val ok = try {
            shell?.injectKey(displayId, keyCode) ?: false
        } catch (e: Exception) {
            false
        }
        if (!ok) logInjectFailureOnce()
    }

    override fun close(packageName: String) {
        sh("am force-stop $packageName")
    }

    private fun findTaskId(packageName: String): Int {
        val dump = sh("dumpsys activity recents", quiet = true)
        val pattern = Regex("""Task\{\S+ #(\d+)[^}]*?A=(?:\d+:)?${Regex.escape(packageName)}\b""")
        return pattern.find(dump)?.groupValues?.get(1)?.toIntOrNull() ?: -1
    }

    private fun logInjectFailureOnce() {
        if (injectFailureLogged) return
        injectFailureLogged = true
        DiagLog.log(TAG, "Input injection failed. Touch and keys will not reach the embedded app")
    }

    private fun sh(command: String, quiet: Boolean = false): String {
        val service = shell
        if (service == null) {
            DiagLog.log(TAG, "$command -> no shell connected")
            return ""
        }
        val output = try {
            service.exec(command)
        } catch (e: Exception) {
            "[remote error] ${e.message}"
        }
        if (!quiet) {
            val trimmed = output.trim().take(LOG_OUTPUT_LIMIT)
            DiagLog.log(TAG, if (trimmed.isEmpty()) command else "$command -> $trimmed")
        }
        return output
    }
}

    private fun openBubble(pkg: String) {
        if (!ready) {
            if (pkg !in pendingAdds) pendingAdds.add(pkg)
            return

        val change = manager.open(pkg)
        change.evicted.forEach { windows.remove(it)?.release() }
        change.collapsed.forEach { windows[it]?.hide() }
        windows.getOrPut(pkg) { createWindow(pkg) }.show()
        topWindow = pkg
        if (arranging) arrangeWindows()
    }

    private fun minimizeBubble(pkg: String) {
        manager.collapse(pkg)
        windows[pkg]?.hide()
        if (arranging) arrangeWindows()
    }

    private fun closeBubble(pkg: String) {
        manager.dismiss(pkg)
        windows.remove(pkg)?.release()
        if (arranging) arrangeWindows()
    }

    private fun onBubbleTap(pkg: String) {
        val open = manager.snapshot().firstOrNull { it.pkg == pkg }?.expanded == true
        if (open) minimizeBubble(pkg) else openBubble(pkg)
    }

    private fun onEmbeddedFailed(pkg: String) {
        embeddedBroken = true
        windows.remove(pkg)?.release()
        ui.postDelayed({
            if (manager.snapshot().any { it.pkg == pkg }) openBubble(pkg)
        }, 400)
    }

    private fun raiseWindow(pkg: String) {
        if (topWindow == pkg) return
        windows[pkg]?.raise()
        topWindow = pkg
    }

    // ---- arrange mode: tile the open windows neatly, in bubble order

    private fun openPackages() = manager.snapshot().filter { it.expanded }.map { it.pkg }

    private fun toggleArrange() = setArranging(!arranging)

    private fun setArranging(on: Boolean) {
        if (on && openPackages().isEmpty()) {
            DiagLog.log(TAG, "Open at least one window before arranging")
            return
        }
        arranging = on
        styleChip()
        DiagLog.log(TAG, if (on) "Arrange mode on. Drag a window to reorder, long press the chip to leave" else "Arrange mode off")
        if (on) arrangeWindows()
    }

    private fun arrangeWindows() {
        val open = openPackages()
        if (open.isEmpty()) {
            setArranging(false)
            return
        }
        val slots = slotRects(open.size)
        open.forEachIndexed { index, pkg -> windows[pkg]?.setBounds(slots[index], animate = true) }
    }

    /** A dropped window takes the nearest slot and the others shuffle to keep a neat order. */
    private fun onWindowDragEnd(pkg: String, centerX: Int, centerY: Int) {
        raiseWindow(pkg)
        if (!arranging) return
        val open = openPackages()
        if (pkg !in open) return
        val slots = slotRects(open.size)
        val nearest = slots.indices.minByOrNull { index ->
            hypot((slots[index].centerX() - centerX).toDouble(), (slots[index].centerY() - centerY).toDouble())
        } ?: return
        val targetPkg = open[nearest]
        if (targetPkg != pkg) {
            val targetIndex = manager.snapshot().indexOfFirst { it.pkg == targetPkg }
            manager.moveTo(pkg, targetIndex)
        }
        arrangeWindows()
    }

    /** One window fills the area. Two stack in portrait or sit side by side in landscape. */
    private fun slotRects(count: Int): List<Rect> {
        if (count <= 0) return emptyList()
        val metrics = resources.displayMetrics
        val gap = dp(8)
        val area = Rect(gap, dp(36), metrics.widthPixels - dp(BUBBLE_DP) - gap, metrics.heightPixels - dp(24))
        val portrait = metrics.heightPixels >= metrics.widthPixels
        val columns = if (count == 1 || (count == 2 && portrait)) 1 else 2
        val rows = (count + columns - 1) / columns
        val cellWidth = (area.width() - gap * (columns - 1)) / columns
        val cellHeight = (area.height() - gap * (rows - 1)) / rows
        return List(count) { index ->
            val row = index / columns
            val column = index % columns
            val alone = columns > 1 && index == count - 1 && count % columns == 1
            val left = area.left + column * (cellWidth + gap)
            val top = area.top + row * (cellHeight + gap)
            Rect(left, top, left + if (alone) area.width() else cellWidth, top + cellHeight)
        }
    }

    // ---- rendering (main thread only)

    private fun overlayParams(width: Int, height: Int, x: Int, y: Int) = WindowManager.LayoutParams(
        width, height,
        WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
        WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
        PixelFormat.TRANSLUCENT
    ).apply {
        gravity = Gravity.TOP or Gravity.START
        this.x = x
        this.y = y
    }

    /** Updates views in place so a dragged bubble keeps its position when state changes. */
    private fun render() {
        val current = manager.snapshot()
        val live = current.map { it.pkg }.toSet()
        handles.keys.filter { it !in live }.forEach { removeHandle(it) }

        current.forEachIndexed { index, bubble ->
            val handle = handles[bubble.pkg]
                ?: createBubble(bubble.pkg, index)?.also { handles[bubble.pkg] = it }
                ?: return@forEachIndexed
            handle.icon.setBackgroundResource(
                if (bubble.expanded) R.drawable.bg_bubble_active else R.drawable.bg_bubble
            )
        }
        ensureChip()
    }

    private fun positionFor(key: String, fallback: Point): Point = positions.getOrPut(key) {
        val saved = positionPrefs.getString(key, null)?.split(",")
        val x = saved?.getOrNull(0)?.toIntOrNull()
        val y = saved?.getOrNull(1)?.toIntOrNull()
        if (x != null && y != null) clampToScreen(Point(x, y)) else fallback
    }

    private fun clampToScreen(point: Point): Point {
        val metrics = resources.displayMetrics
        val size = dp(BUBBLE_DP)
        return Point(point.x.coerceIn(0, metrics.widthPixels - size), point.y.coerceIn(0, metrics.heightPixels - size))
    }

    private fun savePosition(key: String, point: Point) {
        positions[key] = point
        positionPrefs.edit().putString(key, "${point.x},${point.y}").apply()
    }

    private fun createBubble(pkg: String, index: Int): BubbleHandle? {

        val size = dp(BUBBLE_DP)
        val screen = resources.displayMetrics

        val params = overlayParams(size, size, position.x, position.y)

        }
    }

    private fun removeHandle(pkg: String) {
        handles.remove(pkg)?.let { popOutAndRemove(it.root) }
        positions.remove(pkg)
        positionPrefs.edit().remove(pkg).apply()
    }

    private fun labelOf(pkg: String): String = runCatching {

    // ---- animations

    private fun popIn(view: View) {
        view.scaleX = 0f
        view.scaleY = 0f
        view.post {
            view.animate().scaleX(1f).scaleY(1f).setDuration(260)
                .setInterpolator(OvershootInterpolator(1.4f)).start()
        }
    }

    private fun popOutAndRemove(view: View) {
        view.animate().scaleX(0f).scaleY(0f).alpha(0f).setDuration(160)
            .withEndAction { runCatching { wm.removeView(view) } }.start()
    }

    // ---- drag, tap, long press and drop-to-close

    private fun attachDrag(
        view: View,
        params: WindowManager.LayoutParams,
        key: String,
        closable: Boolean,
        onTap: () -> Unit,
        onLongPress: (() -> Unit)? = null
    ) {
        val slop = ViewConfiguration.get(this).scaledTouchSlop
        var startX = 0
        var startY = 0
        var downX = 0f
        var downY = 0f
        var dragging = false
        var longPressed = false
        val longPressCheck = Runnable {
            if (!dragging) {
                longPressed = true
                view.performHapticFeedback(HapticFeedbackConstants.LONG_PRESS)
                onLongPress?.invoke()
            }
        }

        view.setOnTouchListener { _, event ->
            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    startX = params.x
                    startY = params.y
                    downX = event.rawX
                    downY = event.rawY
                    dragging = false
                    longPressed = false
                    if (onLongPress != null) {
                        ui.postDelayed(longPressCheck, ViewConfiguration.getLongPressTimeout().toLong())
                    }
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val dx = event.rawX - downX
                    val dy = event.rawY - downY
                    if (!dragging && !longPressed && hypot(dx, dy) > slop) {
                        dragging = true
                        ui.removeCallbacks(longPressCheck)
                        if (closable) showCloseZone()
                    }
                    if (dragging) {
                        params.x = startX + dx.toInt()
                        params.y = startY + dy.toInt()
                        runCatching { wm.updateViewLayout(view, params) }
                        if (closable) setCloseHot(isOverClose(event.rawX, event.rawY))
                    }
                    true
                }
                MotionEvent.ACTION_UP -> {
                    ui.removeCallbacks(longPressCheck)
                    when {
                        longPressed -> Unit
                        !dragging -> onTap()
                        else -> {
                            val dropped = closable && isOverClose(event.rawX, event.rawY)
                            hideCloseZone()
                            if (dropped) closeBubble(key) else snapToEdge(view, params, key)
                        }
                    }
                    true
                }
                MotionEvent.ACTION_CANCEL -> {
                    ui.removeCallbacks(longPressCheck)
                    hideCloseZone()
                    true
                }
                else -> false
            }
        }
    }

    private fun snapToEdge(view: View, params: WindowManager.LayoutParams, key: String) {
        val screen = resources.displayMetrics
        val targetX = if (params.x + params.width / 2 < screen.widthPixels / 2) 0 else screen.widthPixels - params.width
        val targetY = params.y.coerceIn(0, screen.heightPixels - params.height)
        savePosition(key, Point(targetX, targetY))
        val fromX = params.x
        val fromY = params.y
        ValueAnimator.ofFloat(0f, 1f).apply {
            duration = 180
            interpolator = DecelerateInterpolator()
            addUpdateListener {
                val t = it.animatedValue as Float
                params.x = (fromX + (targetX - fromX) * t).toInt()
                params.y = (fromY + (targetY - fromY) * t).toInt()
                runCatching { wm.updateViewLayout(view, params) }
            }
            start()
        }
    }

    private fun closeZoneTop(): Int =
        resources.displayMetrics.heightPixels - dp(CLOSE_ZONE_DP) - dp(CLOSE_ZONE_BOTTOM_DP)

    private fun isOverClose(rawX: Float, rawY: Float): Boolean {
        val centerX = resources.displayMetrics.widthPixels / 2f
        val centerY = closeZoneTop() + dp(CLOSE_ZONE_DP) / 2f
        return hypot(rawX - centerX, rawY - centerY) < dp(CLOSE_HIT_RADIUS_DP)
    }

    private fun showCloseZone() {
        if (closeZone != null) return
        val view = LayoutInflater.from(themed).inflate(R.layout.view_close_zone, null)
        val size = dp(CLOSE_ZONE_DP)
        val x = (resources.displayMetrics.widthPixels - size) / 2
        view.alpha = 0f
        view.scaleX = 0.6f
        view.scaleY = 0.6f
        runCatching { wm.addView(view, overlayParams(size, size, x, closeZoneTop())) }
            .onSuccess {
                closeZone = view
                view.post { view.animate().alpha(1f).scaleX(1f).scaleY(1f).setDuration(180).start() }
            }
    }

    private fun setCloseHot(hot: Boolean) {
        val zone = closeZone ?: return
        zone.findViewById<View>(R.id.close_bg).setBackgroundResource(
            if (hot) R.drawable.bg_close_zone_hot else R.drawable.bg_close_zone
        )
        val iconAttr = if (hot) {
            com.google.android.material.R.attr.colorOnError
        } else {
            com.google.android.material.R.attr.colorOnErrorContainer
        }
        zone.findViewById<ImageView>(R.id.close_icon).imageTintList =
            ColorStateList.valueOf(MaterialColors.getColor(themed, iconAttr, 0))
        val scale = if (hot) 1.2f else 1f
        zone.animate().scaleX(scale).scaleY(scale).setDuration(100).start()
    }

    private fun hideCloseZone() {
        val zone = closeZone ?: return
        closeZone = null
        zone.animate().alpha(0f).scaleX(0.6f).scaleY(0.6f).setDuration(120)
            .withEndAction { runCatching { wm.removeView(zone) } }.start()
    }

    // ---- chip: movable app drawer button, long press arranges windows

    private fun ensureChip() {
        if (chip != null) return
        val root = LayoutInflater.from(themed).inflate(R.layout.view_bubble, null)
        val icon = root.findViewById<ImageView>(R.id.bubble_icon)
        icon.setImageResource(R.drawable.ic_apps)
        root.contentDescription = getString(R.string.cd_drawer)

        val size = dp(BUBBLE_DP)
        val screen = resources.displayMetrics
        val position = positionFor(
            CHIP_KEY,
            Point(screen.widthPixels - size, dp(FIRST_SLOT_TOP_DP) + BubbleManager.MAX_BUBBLES * dp(SLOT_STEP_DP))
        )
        val params = overlayParams(size, size, position.x, position.y)
        attachDrag(
            root, params, CHIP_KEY, closable = false,
            onTap = { toggleDrawer() },
            onLongPress = { toggleArrange() }
        )
        try {
            popIn(root)
            wm.addView(root, params)
            chip = BubbleHandle(root, icon, params)
            styleChip()
        } catch (e: Exception) {
            DiagLog.log(TAG, "Could not show the chip: ${e.message}")
        }
    }

    private fun styleChip() {
        val icon = chip?.icon ?: return
        icon.setBackgroundResource(if (arranging) R.drawable.bg_chip_active else R.drawable.bg_chip)
        val attr = if (arranging) {
            com.google.android.material.R.attr.colorOnPrimary
        } else {
            com.google.android.material.R.attr.colorOnSecondaryContainer
        }
        icon.imageTintList = ColorStateList.valueOf(MaterialColors.getColor(themed, attr, 0))
    }

    // ---- app drawer (our own overlay, because a service cannot show a bottom sheet)

    private fun toggleDrawer() = if (drawer == null) showDrawer() else hideDrawer()

    private fun showDrawer() {
        val view = LayoutInflater.from(themed).inflate(R.layout.view_app_drawer, null)
        val adapter = AppListAdapter { entry ->
            hideDrawer()
            openBubble(entry.packageName)
        }
        view.findViewById<RecyclerView>(R.id.list).apply {
            layoutManager = LinearLayoutManager(themed)
            this.adapter = adapter
        }
        view.findViewById<EditText>(R.id.search)
            .doAfterTextChanged { adapter.filter(it?.toString().orEmpty()) }
        view.setOnTouchListener { _, event ->
            if (event.action == MotionEvent.ACTION_OUTSIDE) {
                hideDrawer()
                true
            } else {
                false
            }
        }

        val screen = resources.displayMetrics
        val params = WindowManager.LayoutParams(
            minOf(dp(DRAWER_MAX_WIDTH_DP), (screen.widthPixels * 0.92f).toInt()),
            (screen.heightPixels * 0.62f).toInt(),
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            // Focusable so the search field can open the keyboard.
            WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL or WindowManager.LayoutParams.FLAG_WATCH_OUTSIDE_TOUCH,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.CENTER
            softInputMode = WindowManager.LayoutParams.SOFT_INPUT_ADJUST_PAN
        }
        view.alpha = 0f
        view.translationY = dp(24).toFloat()
        try {
            wm.addView(view, params)
        } catch (e: Exception) {
            DiagLog.log(TAG, "Could not open the app drawer: ${e.message}")
            return
        }
        drawer = view
        view.post {
            view.animate().alpha(1f).translationY(0f).setDuration(220)
                .setInterpolator(DecelerateInterpolator()).start()
        }

        val recents = manager.recentsSnapshot()
        thread {
            val apps = AppList.load(this, recents)
            ui.post { adapter.submit(apps) }
        }
    }

    private fun hideDrawer() {
        val view = drawer ?: return
        drawer = null
        view.animate().alpha(0f).translationY(dp(24).toFloat()).setDuration(160)
            .withEndAction { runCatching { wm.removeView(view) } }.start()
    }
}

package com.multitask.bubble.ui

import android.content.Context
import android.content.Intent
import android.graphics.drawable.Drawable
import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.RecyclerView
import com.multitask.bubble.R
import com.multitask.bubble.databinding.ItemAppBinding

class AppEntry(
    val packageName: String,
    val label: String,
    val icon: Drawable,
    val recent: Boolean
)

object AppList {
    /** Every launchable app. Recent bubbles come first, the rest are alphabetical. Blocking. */
    fun load(context: Context, recents: List<String>): List<AppEntry> {
        val pm = context.packageManager
        val launcher = Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER)
        val all = pm.queryIntentActivities(launcher, 0)
            .filter { it.activityInfo.packageName != context.packageName }
            .distinctBy { it.activityInfo.packageName }
            .map {
                val pkg = it.activityInfo.packageName
                AppEntry(pkg, it.loadLabel(pm).toString(), it.loadIcon(pm), pkg in recents)
            }
        val recentFirst = recents.mapNotNull { pkg -> all.firstOrNull { it.packageName == pkg } }
        val rest = all.filter { !it.recent }.sortedBy { it.label.lowercase() }
        return recentFirst + rest
    }
}

class AppListAdapter(
    private val onClick: (AppEntry) -> Unit
) : RecyclerView.Adapter<AppListAdapter.Holder>() {

    class Holder(val binding: ItemAppBinding) : RecyclerView.ViewHolder(binding.root)

    private var all: List<AppEntry> = emptyList()
    private var shown: List<AppEntry> = emptyList()
    private var query = ""

    fun submit(items: List<AppEntry>) {
        all = items
        filter(query)
    }

    fun filter(text: String) {
        query = text
        shown = if (text.isBlank()) all else all.filter { it.label.contains(text, ignoreCase = true) }
        notifyDataSetChanged()
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int) =
        Holder(ItemAppBinding.inflate(LayoutInflater.from(parent.context), parent, false))

    override fun getItemCount() = shown.size

    override fun onBindViewHolder(holder: Holder, position: Int) {
        val entry = shown[position]
        val context = holder.binding.root.context
        holder.binding.appIcon.setImageDrawable(entry.icon)
        holder.binding.appLabel.text = entry.label
        holder.binding.appPackage.text =
            if (entry.recent) context.getString(R.string.recent_label) else entry.packageName
        holder.binding.root.setOnClickListener { onClick(entry) }
    }
}

package com.multitask.bubble.ui

import android.graphics.Rect

/** The window an app lives in while it is a bubble. Main thread only. */
interface BubbleWindow {
    /** Brings the app on screen. */
    fun show()

    /** Takes the app off screen but keeps it running. */
    fun hide()

    /** Closes the app and frees everything. */
    fun release()

    /** Places and sizes the window. Applied on the next [show] if the window is hidden. */
    fun setBounds(bounds: Rect, animate: Boolean) {}

    /** Puts the window above the other windows. */
    fun raise() {}
}

package com.multitask.bubble.ui

import android.animation.ValueAnimator
import android.content.Context
import android.graphics.PixelFormat
import android.graphics.Point
import android.graphics.Rect
import android.hardware.display.DisplayManager
import android.hardware.display.VirtualDisplay
import android.os.Handler
import android.os.Looper
import android.os.SystemClock
import android.view.Gravity
import android.view.KeyEvent
import android.view.LayoutInflater
import android.view.MotionEvent
import android.view.SurfaceHolder
import android.view.SurfaceView
import android.view.View
import android.view.ViewConfiguration
import android.view.WindowManager
import android.view.animation.DecelerateInterpolator
import android.widget.ImageView
import android.widget.TextView
import androidx.core.animation.doOnEnd
import com.multitask.bubble.R
import com.multitask.bubble.core.DiagLog
import com.multitask.bubble.core.WindowBackend
import java.util.concurrent.Executor
import java.util.concurrent.Executors
import kotlin.math.abs

/**
 * The real bubble: the app runs on a private virtual display and its pixels are drawn into a
 * SurfaceView inside our own overlay window. We own the frame, so minimize, close, move and
 * resize are real. Needs only the public DisplayManager API plus Shizuku for launching the app
 * and injecting input.
 *
 * [onFailed] fires once if the device cannot run apps on a secondary display, so the caller can
 * switch to the freeform fallback. [onFocus] fires when the title bar is tapped. [onDragEnd]
 * fires with the window center when the title bar was dragged and released.
 */
class EmbeddedWindow(
    private val context: Context,
    private val wm: WindowManager,
    private val backend: WindowBackend,
    private val worker: Executor,
    private val pkg: String,
    private val onMinimize: () -> Unit,
    private val onClose: () -> Unit,
    private val onFailed: () -> Unit,
    private val onFocus: () -> Unit,
    private val onDragEnd: (centerX: Int, centerY: Int) -> Unit
) : BubbleWindow, SurfaceHolder.Callback {

    private companion object {
        const val TAG = "EmbeddedWindow"
        const val FLAG_SUPPORTS_TOUCH = 1 shl 6 // hidden DisplayManager flag
        const val WIDTH_FRACTION = 0.8f
        const val HEIGHT_FRACTION = 0.6f
        const val TITLE_DP = 52
        const val MIN_SURFACE_DP = 120
        const val SLIDE_DP = 16
        const val ENTER_MS = 220L
        const val EXIT_MS = 150L
        const val MOVE_MS = 280L
    }

    private val ui = Handler(Looper.getMainLooper())
    private val metrics = context.resources.displayMetrics
    private fun dp(value: Int) = (value * metrics.density).toInt()

    private var surfaceWidth = (metrics.widthPixels * WIDTH_FRACTION).toInt()
    private var surfaceHeight = (metrics.heightPixels * HEIGHT_FRACTION).toInt()
    private val position = Point((metrics.widthPixels * 0.04f).toInt(), (metrics.heightPixels * 0.1f).toInt())
    private val input = Executors.newSingleThreadExecutor()

    private var root: View? = null
    private var windowParams: WindowManager.LayoutParams? = null
    private var activeHolder: SurfaceHolder? = null
    private var virtualDisplay: VirtualDisplay? = null
    private var displayId = -1
    private var downTime = 0L
    private var released = false
    private var failed = false
    private var enterAnimator: ValueAnimator? = null
    private var moveAnimator: ValueAnimator? = null

    // ---- BubbleWindow

    override fun show() = attach(animated = true)

    override fun hide() = detach(animated = true)

    override fun release() {
        released = true
        detach(animated = true)
        val display = virtualDisplay
        virtualDisplay = null
        displayId = -1
        input.shutdown()
        worker.execute {
            // Stop the app first, otherwise Android moves it to the main screen when the display goes.
            backend.close(pkg)
            Thread.sleep(250)
            display?.release()
        }
    }

    override fun raise() {
        if (root == null || released) return
        detach(animated = false)
        attach(animated = false)
    }

    override fun setBounds(bounds: Rect, animate: Boolean) {
        surfaceWidth = bounds.width().coerceAtLeast(dp(MIN_SURFACE_DP))
        surfaceHeight = (bounds.height() - dp(TITLE_DP)).coerceAtLeast(dp(MIN_SURFACE_DP))
        virtualDisplay?.resize(surfaceWidth, surfaceHeight, metrics.densityDpi)

        val view = root
        val params = windowParams
        if (view == null || params == null) {
            position.set(bounds.left, bounds.top)
            return
        }
        val surface = view.findViewById<SurfaceView>(R.id.surface)
        surface.layoutParams = surface.layoutParams.apply {
            width = surfaceWidth
            height = surfaceHeight
        }
        moveAnimator?.cancel()
        if (!animate) {
            params.x = bounds.left
            params.y = bounds.top
            position.set(params.x, params.y)
            runCatching { wm.updateViewLayout(view, params) }
            return
        }
        val fromX = params.x
        val fromY = params.y
        moveAnimator = ValueAnimator.ofFloat(0f, 1f).apply {
            duration = MOVE_MS
            interpolator = DecelerateInterpolator(1.5f)
            addUpdateListener {
                if (root !== view) return@addUpdateListener
                val t = it.animatedValue as Float
                params.x = (fromX + (bounds.left - fromX) * t).toInt()
                params.y = (fromY + (bounds.top - fromY) * t).toInt()
                position.set(params.x, params.y)
                runCatching { wm.updateViewLayout(view, params) }
            }
            start()
        }
    }

    // ---- window

    private fun attach(animated: Boolean) {
        if (root != null || released) return
        val view = LayoutInflater.from(context).inflate(R.layout.view_embedded_window, null)

        val surface = view.findViewById<SurfaceView>(R.id.surface)
        surface.layoutParams = surface.layoutParams.apply {
            width = surfaceWidth
            height = surfaceHeight
        }
        surface.holder.addCallback(this)
        surface.setOnTouchListener { _, event ->
            forward(event)
            true
        }

        val pm = context.packageManager
        view.findViewById<TextView>(R.id.title_text).text = runCatching {
            pm.getApplicationLabel(pm.getApplicationInfo(pkg, 0)).toString()
        }.getOrDefault(pkg)
        view.findViewById<ImageView>(R.id.title_icon)
            .setImageDrawable(runCatching { pm.getApplicationIcon(pkg) }.getOrNull())
        view.findViewById<View>(R.id.btn_back).setOnClickListener { sendKey(KeyEvent.KEYCODE_BACK) }
        view.findViewById<View>(R.id.btn_minimize).setOnClickListener { onMinimize() }
        view.findViewById<View>(R.id.btn_close).setOnClickListener { onClose() }

        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.START
            x = position.x
            y = position.y + if (animated) dp(SLIDE_DP) else 0
            alpha = if (animated) 0f else 1f
        }
        attachTitleDrag(view.findViewById(R.id.title_bar), view, params)

        try {
            wm.addView(view, params)
            root = view
            windowParams = params
        } catch (e: Exception) {
            DiagLog.log(TAG, "Could not show window for $pkg: ${e.message}")
            return
        }
        if (animated) animateEnter(view, params)
    }

    private fun animateEnter(view: View, params: WindowManager.LayoutParams) {
        enterAnimator?.cancel()
        enterAnimator = ValueAnimator.ofFloat(0f, 1f).apply {
            duration = ENTER_MS
            interpolator = DecelerateInterpolator()
            addUpdateListener {
                if (root !== view) return@addUpdateListener
                val t = it.animatedValue as Float
                params.alpha = t
                params.y = position.y + ((1f - t) * dp(SLIDE_DP)).toInt()
                runCatching { wm.updateViewLayout(view, params) }
            }
            start()
        }
    }

    private fun detach(animated: Boolean) {
        val view = root ?: return
        val params = windowParams
        root = null
        windowParams = null
        enterAnimator?.cancel()
        moveAnimator?.cancel()

        fun finish() {
            runCatching { wm.removeView(view) }
            // Pause drawing, unless a new window already took over.
            if (root == null) virtualDisplay?.surface = null
        }
        if (!animated || params == null) {
            finish()
            return
        }
        ValueAnimator.ofFloat(1f, 0f).apply {
            duration = EXIT_MS
            addUpdateListener {
                val t = it.animatedValue as Float
                params.alpha = t
                params.y = position.y + ((1f - t) * dp(SLIDE_DP)).toInt()
                runCatching { wm.updateViewLayout(view, params) }
            }
            doOnEnd { finish() }
            start()
        }
    }

    // ---- surface lifecycle

    override fun surfaceCreated(holder: SurfaceHolder) {
        activeHolder = holder
        val existing = virtualDisplay
        if (existing != null) {
            existing.surface = holder.surface
            return
        }
        val manager = context.getSystemService(DisplayManager::class.java)
        val base = DisplayManager.VIRTUAL_DISPLAY_FLAG_PUBLIC or DisplayManager.VIRTUAL_DISPLAY_FLAG_OWN_CONTENT_ONLY
        val created = runCatching {
            manager.createVirtualDisplay(
                "bubble-$pkg", surfaceWidth, surfaceHeight, metrics.densityDpi,
                holder.surface, base or FLAG_SUPPORTS_TOUCH
            )
        }.recoverCatching {
            manager.createVirtualDisplay(
                "bubble-$pkg", surfaceWidth, surfaceHeight, metrics.densityDpi, holder.surface, base
            )
        }.onFailure { DiagLog.log(TAG, "Virtual display failed: ${it.message}") }.getOrNull()

        if (created == null) {
            fail()
            return
        }
        virtualDisplay = created
        displayId = created.display.displayId
        DiagLog.log(TAG, "$pkg runs on virtual display $displayId (${surfaceWidth}x$surfaceHeight)")
        val id = displayId
        worker.execute {
            if (!backend.launchOnDisplay(pkg, id)) ui.post { fail() }
        }
    }

    override fun surfaceChanged(holder: SurfaceHolder, format: Int, width: Int, height: Int) = Unit

    override fun surfaceDestroyed(holder: SurfaceHolder) {
        // An old window can be destroyed after a new one took over, so only react to the active one.
        if (holder === activeHolder) {
            activeHolder = null
            virtualDisplay?.surface = null
        }
    }

    private fun fail() {
        if (failed || released) return
        failed = true
        DiagLog.log(TAG, "Embedding is not supported here, switching to freeform windows")
        onFailed()
    }

    // ---- input

    /** Forwards every pointer, so pinch-to-zoom, double tap and flings reach the app unchanged. */
    private fun forward(event: MotionEvent) {
        val id = displayId
        if (id <= 0) return
        when (event.actionMasked) {
            MotionEvent.ACTION_DOWN -> downTime = SystemClock.uptimeMillis()
            MotionEvent.ACTION_MOVE, MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL,
            MotionEvent.ACTION_POINTER_DOWN, MotionEvent.ACTION_POINTER_UP -> Unit
            else -> return
        }
        val count = event.pointerCount
        val ids = IntArray(count) { event.getPointerId(it) }
        val xs = FloatArray(count) { event.getX(it) }
        val ys = FloatArray(count) { event.getY(it) }
        val action = event.action // keeps the pointer index for POINTER_DOWN/UP
        val time = downTime
        runCatching { input.execute { backend.injectTouch(id, action, time, ids, xs, ys) } }
    }

    private fun sendKey(keyCode: Int) {
        val id = displayId
        if (id <= 0) return
        runCatching { input.execute { backend.injectKey(id, keyCode) } }
    }

    private fun attachTitleDrag(handle: View, window: View, params: WindowManager.LayoutParams) {
        val slop = ViewConfiguration.get(context).scaledTouchSlop
        var startX = 0
        var startY = 0
        var downX = 0f
        var downY = 0f
        var moved = false
        handle.setOnTouchListener { _, event ->
            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    moveAnimator?.cancel()
                    startX = params.x
                    startY = params.y
                    downX = event.rawX
                    downY = event.rawY
                    moved = false
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val dx = (event.rawX - downX).toInt()
                    val dy = (event.rawY - downY).toInt()
                    if (moved || abs(dx) > slop || abs(dy) > slop) {
                        moved = true
                        params.x = startX + dx
                        params.y = startY + dy
                        position.set(params.x, params.y)
                        runCatching { wm.updateViewLayout(window, params) }
                    }
                    true
                }
                MotionEvent.ACTION_UP -> {
                    if (moved) {
                        onDragEnd(params.x + surfaceWidth / 2, params.y + (surfaceHeight + dp(TITLE_DP)) / 2)
                    } else {
                        onFocus()
                    }
                    true
                }
                else -> true
            }
        }
    }
}

===== FILE: app/src/main/java/com/multitask/bubble/ui/FreeformWindow.kt =====
package com.multitask.bubble.ui

import android.content.Context
import android.graphics.Rect
import com.multitask.bubble.core.WindowBackend
import java.util.concurrent.Executor

/**
 * Fallback for devices that cannot run apps on a secondary display. The system draws the window,
 * we only place it. Hiding moves it off screen, which some ROMs may ignore.
 */
class FreeformWindow(
    context: Context,
    private val backend: WindowBackend,
    private val worker: Executor,
    private val pkg: String
) : BubbleWindow {

    private val offscreen = Rect(-3000, -3000, -2000, -2000)

    @Volatile
    private var bounds: Rect

    @Volatile
    private var taskId = -1

    init {
        val metrics = context.resources.displayMetrics
        val w = metrics.widthPixels
        val h = metrics.heightPixels
        bounds = Rect((w * 0.08).toInt(), (h * 0.12).toInt(), (w * 0.92).toInt(), (h * 0.72).toInt())
    }

    override fun show() {
        val target = bounds
        // Launching again reuses the existing task and brings it to the front.
        worker.execute {
            val id = backend.launchFreeform(pkg, target)
            if (id > 0) taskId = id
        }
    }

    override fun hide() {
        worker.execute { if (taskId > 0) backend.moveTask(taskId, offscreen) }
    }

    override fun release() {
        worker.execute { backend.close(pkg) }
    }

    override fun setBounds(bounds: Rect, animate: Boolean) {
        this.bounds = Rect(bounds)
        val target = this.bounds
        worker.execute { if (taskId > 0) backend.moveTask(taskId, target) }
    }
}

===== FILE: app/src/test/java/com/multitask/bubble/BubbleManagerTest.kt =====
package com.multitask.bubble

import android.content.Context
import androidx.test.core.app.ApplicationProvider
import com.multitask.bubble.core.BubbleManager
import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Test
import org.junit.runner.RunWith
import org.robolectric.RobolectricTestRunner
import org.robolectric.annotation.Config

@RunWith(RobolectricTestRunner::class)
@Config(sdk = [34])
class BubbleManagerTest {

    private val context: Context get() = ApplicationProvider.getApplicationContext()

    @Test
    fun capEvictsTheOldestBubble() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")
        manager.open("app.two")
        manager.open("app.three")
        assertEquals(BubbleManager.MAX_BUBBLES, manager.snapshot().size)

        val change = manager.open("app.four")

        assertEquals(listOf("app.one"), change.evicted)
        val pkgs = manager.snapshot().map { it.pkg }
        assertFalse(pkgs.contains("app.one"))
        assertEquals("app.one", manager.recentsSnapshot().first())
    }

    @Test
    fun severalBubblesCanBeOpenByDefault() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")

        val change = manager.open("app.two")

        assertTrue(change.collapsed.isEmpty())
        assertTrue(manager.snapshot().all { it.expanded })
    }

    @Test
    fun singleOpenModeCollapsesTheOthers() {
        val manager = BubbleManager(context, singleOpen = true) {}
        manager.open("app.one")

        val change = manager.open("app.two")

        assertEquals(listOf("app.one"), change.collapsed)
        val state = manager.snapshot().associateBy { it.pkg }
        assertFalse(state.getValue("app.one").expanded)
        assertTrue(state.getValue("app.two").expanded)
    }

    @Test
    fun openingAnExistingAppDoesNotDuplicateIt() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")
        manager.open("app.one")
        assertEquals(1, manager.snapshot().size)
    }

    @Test
    fun collapseKeepsTheBubbleButClosesIt() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")

        manager.collapse("app.one")

        assertEquals(1, manager.snapshot().size)
        assertFalse(manager.snapshot().first().expanded)
    }

    @Test
    fun dismissedBubblesMoveToRecentsAndReopen() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")
        manager.dismiss("app.one")
        assertTrue(manager.snapshot().isEmpty())
        assertEquals(listOf("app.one"), manager.recentsSnapshot())

        manager.open("app.one")
        assertEquals(1, manager.snapshot().size)
        assertTrue(manager.recentsSnapshot().isEmpty())
    }

    @Test
    fun bubblesAreRestoredCollapsedByANewManager() {
        BubbleManager(context) {}.open("app.one")

        val restored = BubbleManager(context) {}.snapshot()

        assertEquals(listOf("app.one"), restored.map { it.pkg })
        assertFalse(restored.first().expanded)
    }

    @Test
    fun moveToPutsTheBubbleAtTheGivenIndex() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")
        manager.open("app.two")
        manager.open("app.three")

        manager.moveTo("app.three", 0)
        assertEquals(listOf("app.three", "app.one", "app.two"), manager.snapshot().map { it.pkg })

        manager.moveTo("app.three", 2)
        assertEquals(listOf("app.one", "app.two", "app.three"), manager.snapshot().map { it.pkg })
    }

    @Test
    fun changesNotifyTheListener() {
        var calls = 0
        val manager = BubbleManager(context) { calls++ }
        manager.open("app.one")
        manager.collapse("app.one")
        manager.dismiss("app.one")
        assertEquals(3, calls)
    }
}
