package com.multitask.bubble.core

import android.content.Context
import android.graphics.Rect

/**
 * Owns the bubble list.
 *
 * Mutating methods ([add], [toggle], [dismiss]) block on shell commands, so call them from a single
 * background thread, one at a time. [snapshot] and [recentsSnapshot] never block and are safe to
 * read from the main thread. [onChanged] fires after every change, on the calling thread.
 */
class BubbleManager(
    context: Context,
    private val backend: WindowBackend,
    private val onChanged: () -> Unit
) {
    data class Bubble(val pkg: String, val taskId: Int = -1, val expanded: Boolean = false)

    companion object {
        const val MAX_BUBBLES = 3
        const val MAX_RECENTS = 8
        private const val PREFS = "bubbles"
        private const val KEY_ACTIVE = "active"
        private const val KEY_RECENTS = "recents"
        private val OFFSCREEN = Rect(-3000, -3000, -2000, -2000)
    }

    private val appContext = context.applicationContext
    private val prefs = appContext.getSharedPreferences(PREFS, Context.MODE_PRIVATE)

    private val bubbles = mutableListOf<Bubble>()
    private val recents = mutableListOf<String>()

    @Volatile
    private var publishedBubbles: List<Bubble> = emptyList()

    @Volatile
    private var publishedRecents: List<String> = emptyList()

    init {
        restore()
        publish()
    }

    fun snapshot(): List<Bubble> = publishedBubbles

    fun recentsSnapshot(): List<String> = publishedRecents

    fun add(pkg: String) {
        if (bubbles.none { it.pkg == pkg }) {
            if (bubbles.size >= MAX_BUBBLES) evict(bubbles.first().pkg)
            bubbles.add(Bubble(pkg))
            recents.remove(pkg)
        }
        toggle(pkg)
    }

    /** Opens the bubble if it is collapsed, collapses it if it is open. Only one is open at a time. */
    fun toggle(pkg: String) {
        val target = bubbles.find { it.pkg == pkg } ?: return
        if (target.expanded) {
            collapse(target)
        } else {
            bubbles.filter { it.expanded }.forEach { collapse(it) }
            expand(target)
        }
        commit()
    }

    fun dismiss(pkg: String) {
        if (bubbles.none { it.pkg == pkg }) return
        evict(pkg)
        commit()
    }

    private fun expand(bubble: Bubble) {
        // Launching again reuses the existing task and brings it to the front.
        val taskId = backend.launch(bubble.pkg, windowBounds())
        replace(bubble.pkg) { it.copy(taskId = if (taskId > 0) taskId else it.taskId, expanded = true) }
    }

    private fun collapse(bubble: Bubble) {
        if (bubble.taskId > 0) backend.moveTask(bubble.taskId, OFFSCREEN)
        replace(bubble.pkg) { it.copy(expanded = false) }
    }

    private fun evict(pkg: String) {
        backend.close(pkg)
        bubbles.removeAll { it.pkg == pkg }
        recents.remove(pkg)
        recents.add(0, pkg)
        while (recents.size > MAX_RECENTS) recents.removeAt(recents.lastIndex)
    }

    private fun replace(pkg: String, change: (Bubble) -> Bubble) {
        val index = bubbles.indexOfFirst { it.pkg == pkg }
        if (index >= 0) bubbles[index] = change(bubbles[index])
    }

    private fun windowBounds(): Rect {
        val metrics = appContext.resources.displayMetrics
        val w = metrics.widthPixels
        val h = metrics.heightPixels
        return Rect((w * 0.08).toInt(), (h * 0.12).toInt(), (w * 0.92).toInt(), (h * 0.72).toInt())
    }

    private fun commit() {
        save()
        publish()
        onChanged()
    }

    private fun publish() {
        publishedBubbles = bubbles.toList()
        publishedRecents = recents.toList()
    }

    private fun save() {
        prefs.edit()
            .putString(KEY_ACTIVE, bubbles.joinToString(",") { it.pkg })
            .putString(KEY_RECENTS, recents.joinToString(","))
            .apply()
    }

    /** Restored bubbles start collapsed; their apps are relaunched on first tap. */
    private fun restore() {
        prefs.getString(KEY_ACTIVE, "").orEmpty().split(",").filter { it.isNotBlank() }
            .take(MAX_BUBBLES).forEach { bubbles.add(Bubble(it)) }
        prefs.getString(KEY_RECENTS, "").orEmpty().split(",").filter { it.isNotBlank() }
            .take(MAX_RECENTS).forEach { recents.add(it) }
    }
}

package com.multitask.bubble.core

import android.util.Log
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

/** In-memory log shared by the activity and the overlay service, mirrored to logcat. */
object DiagLog {
    private const val MAX_CHARS = 6000
    private const val TRIM_CHARS = 2000

    private val buffer = StringBuilder()

    /** Called on whichever thread logged. UI code must post to the main thread itself. */
    @Volatile
    var listener: ((String) -> Unit)? = null

    fun log(tag: String, message: String) {
        Log.d(tag, message)
        val stamp = SimpleDateFormat("HH:mm:ss", Locale.getDefault()).format(Date())
        synchronized(buffer) {
            buffer.append('[').append(stamp).append("] ").append(message).append('\n')
            if (buffer.length > MAX_CHARS) buffer.delete(0, TRIM_CHARS)
        }
        listener?.invoke(snapshot())
    }

    fun snapshot(): String = synchronized(buffer) { buffer.toString() }

    fun clear() {
        synchronized(buffer) { buffer.setLength(0) }
        listener?.invoke("")
    }
}

import kotlin.system.exitProcess

/** Runs inside the Shizuku shell process (UID 2000). */
class ShellService : IShellService.Stub() {

    override fun destroy() {
        exitProcess(0)
    }

    override fun exec(command: String): String {
        return try {
            val process = ProcessBuilder("sh", "-c", command)
                .redirectErrorStream(true)
                .start()
            val output = process.inputStream.bufferedReader().readText()
            process.waitFor()
            output
        } catch (e: Exception) {
            "[exception] ${e.message}"
        }
    }
}

package com.multitask.bubble.core

import android.content.ComponentName
import android.content.Context
import android.content.ServiceConnection
import android.content.pm.PackageManager
import android.graphics.Rect
import android.os.IBinder
import rikka.shizuku.Shizuku

class ShizukuWindowBackend(private val context: Context) : WindowBackend {

    companion object {
        const val TAG = "ShizukuBackend"
        private const val REQUEST_CODE = 1001
        private const val WINDOWING_MODE_FREEFORM = 5
        private const val TASK_POLL_ATTEMPTS = 8
        private const val TASK_POLL_DELAY_MS = 250L
        private const val LOG_OUTPUT_LIMIT = 300
    }

    @Volatile
    private var shell: IShellService? = null

    @Volatile
    private var freeformConfigured = false

    private val waiting = mutableListOf<() -> Unit>()

    // Bump version() whenever ShellService changes, or Shizuku keeps running the old process.
    private val userServiceArgs = Shizuku.UserServiceArgs(
        ComponentName(context.packageName, ShellService::class.java.name)
    ).daemon(false).processNameSuffix("shell").debuggable(false).version(2)

    private val connection = object : ServiceConnection {
        override fun onServiceConnected(name: ComponentName?, binder: IBinder?) {
            shell = IShellService.Stub.asInterface(binder)
            DiagLog.log(TAG, "Shell service connected")
            val callbacks = synchronized(waiting) { waiting.toList().also { waiting.clear() } }
            callbacks.forEach { it() }
        }

        override fun onServiceDisconnected(name: ComponentName?) {
            shell = null
            freeformConfigured = false
            DiagLog.log(TAG, "Shell service disconnected")
        }
    }

    override fun isReady(): Boolean = try {
        Shizuku.pingBinder() && Shizuku.checkSelfPermission() == PackageManager.PERMISSION_GRANTED
    } catch (e: Exception) {
        false
    }

    override fun requestPermission(onResult: (Boolean) -> Unit) {
        if (!Shizuku.pingBinder()) {
            onResult(false)
            return
        }
        if (Shizuku.checkSelfPermission() == PackageManager.PERMISSION_GRANTED) {
            onResult(true)
            return
        }
        val listener = object : Shizuku.OnRequestPermissionResultListener {
            override fun onRequestPermissionResult(requestCode: Int, grantResult: Int) {
                if (requestCode != REQUEST_CODE) return
                Shizuku.removeRequestPermissionResultListener(this)
                val granted = grantResult == PackageManager.PERMISSION_GRANTED
                DiagLog.log(TAG, "Shizuku permission ${if (granted) "granted" else "denied"}")
                onResult(granted)
            }
        }
        Shizuku.addRequestPermissionResultListener(listener)
        Shizuku.requestPermission(REQUEST_CODE)
    }

    override fun connect(onConnected: () -> Unit) {
        if (shell != null) {
            onConnected()
            return
        }
        if (!isReady()) {
            DiagLog.log(TAG, "Cannot connect: Shizuku is not running or not authorized")
            return
        }
        val firstWaiter = synchronized(waiting) {
            waiting.add(onConnected)
            waiting.size == 1
        }
        if (firstWaiter) {
            DiagLog.log(TAG, "Binding shell service")
            Shizuku.bindUserService(userServiceArgs, connection)
        }
    }

    override fun disconnect() {
        if (shell == null) return
        runCatching { Shizuku.unbindUserService(userServiceArgs, connection, true) }
        shell = null
        freeformConfigured = false
    }

    override fun setupFreeform() {
        if (freeformConfigured) return
        sh("settings put global enable_freeform_support 1")
        sh("settings put global force_resizable_activities 1")
        freeformConfigured = true
    }

    override fun launch(packageName: String, bounds: Rect): Int {
        val component = context.packageManager.getLaunchIntentForPackage(packageName)
            ?.component?.flattenToShortString()
        if (component == null) {
            DiagLog.log(TAG, "No launcher activity found for $packageName")
            return -1
        }
        sh("am start -n $component --windowingMode $WINDOWING_MODE_FREEFORM")

        var taskId = -1
        for (attempt in 0 until TASK_POLL_ATTEMPTS) {
            taskId = findTaskId(packageName)
            if (taskId > 0) break
            Thread.sleep(TASK_POLL_DELAY_MS)
        }
        if (taskId > 0) {
            DiagLog.log(TAG, "$packageName is task $taskId")
            moveTask(taskId, bounds)
        } else {
            DiagLog.log(TAG, "Could not find a task for $packageName, so it cannot be moved")
        }
        return taskId
    }

    override fun moveTask(taskId: Int, bounds: Rect) {
        sh("am task resize $taskId ${bounds.left},${bounds.top},${bounds.right},${bounds.bottom}")
    }

    override fun close(packageName: String) {
        sh("am force-stop $packageName")
    }

    private fun findTaskId(packageName: String): Int {
        val dump = sh("dumpsys activity recents", quiet = true)
        val pattern = Regex("""Task\{\S+ #(\d+)[^}]*?A=(?:\d+:)?${Regex.escape(packageName)}\b""")
        return pattern.find(dump)?.groupValues?.get(1)?.toIntOrNull() ?: -1
    }

    private fun sh(command: String, quiet: Boolean = false): String {
        val service = shell
        if (service == null) {
            DiagLog.log(TAG, "$command -> no shell connected")
            return ""
        }
        val output = try {
            service.exec(command)
        } catch (e: Exception) {
            "[remote error] ${e.message}"
        }
        if (!quiet) {
            val trimmed = output.trim().take(LOG_OUTPUT_LIMIT)
            DiagLog.log(TAG, if (trimmed.isEmpty()) command else "$command -> $trimmed")
        }
        return output
    }
}

        const val ACTION_ADD = "com.multitask.bubble.ADD"
        const val ACTION_STOP = "com.multitask.bubble.STOP"
        const val EXTRA_PKG = "pkg"

        private const val TAG = "BubbleService"
        private const val CHANNEL_ID = "bubble_service"

        private const val BUBBLE_DP = 56

        private const val CLOSE_ZONE_BOTTOM_DP = 96
        private const val CLOSE_HIT_RADIUS_DP = 56

        private const val SLOT_STEP_DP = 64

        @Volatile
        var running: Boolean = false
            private set

        fun start(context: Context) {
            ContextCompat.startForegroundService(context, Intent(context, BubbleOverlayService::class.java))
        }

        fun add(context: Context, pkg: String) {
            ContextCompat.startForegroundService(
                context,
                Intent(context, BubbleOverlayService::class.java)
                    .setAction(ACTION_ADD)
                    .putExtra(EXTRA_PKG, pkg)
            )
        }

        fun stop(context: Context) {
            context.stopService(Intent(context, BubbleOverlayService::class.java))
        }
    }

    private class BubbleHandle(
        val root: View,
        val icon: ImageView,
        val params: WindowManager.LayoutParams
    )

    private lateinit var themed: Context
    private lateinit var wm: WindowManager
    private lateinit var backend: ShizukuWindowBackend
    private lateinit var manager: BubbleManager

    private val ui = Handler(Looper.getMainLooper())
    private val worker = Executors.newSingleThreadExecutor()
    private val handles = mutableMapOf<String, BubbleHandle>()
    private val positions = mutableMapOf<String, Point>()
    private val pendingAdds = mutableListOf<String>()
    private var ready = false

    private var chip: BubbleHandle? = null
    private var closeZone: View? = null
    private var recentsPanel: View? = null

    private val density by lazy { resources.displayMetrics.density }
    private fun dp(value: Int) = (value * density).toInt()

    override fun onCreate() {
        super.onCreate()
        running = true
        themed = ContextThemeWrapper(this, R.style.Theme_BubbleManager)
        wm = getSystemService(WINDOW_SERVICE) as WindowManager
        backend = ShizukuWindowBackend(applicationContext)
        manager = BubbleManager(applicationContext, backend) { ui.post { render() } }

        startAsForeground()
        render() // bubbles restored from the last session show up right away

        backend.connect {
            worker.execute {
                backend.setupFreeform()
                ui.post {
                    ready = true
                    pendingAdds.forEach { pkg -> worker.execute { manager.add(pkg) } }
                    pendingAdds.clear()
                }
            }
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> stopSelf()
            ACTION_ADD -> intent.getStringExtra(EXTRA_PKG)?.let { pkg ->
                if (ready) worker.execute { manager.add(pkg) } else pendingAdds.add(pkg)
            }
        }
        return START_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onDestroy() {
        running = false
        hideRecents()
        hideCloseZone()
        handles.keys.toList().forEach { removeHandle(it) }
        chip?.let { runCatching { wm.removeView(it.root) } }
        chip = null
        worker.shutdown()
        backend.disconnect()
        super.onDestroy()
    }

    // ---- foreground notification

    private fun startAsForeground() {
        getSystemService(NotificationManager::class.java).createNotificationChannel(
            NotificationChannel(CHANNEL_ID, getString(R.string.notif_channel), NotificationManager.IMPORTANCE_LOW)
        )
        val open = PendingIntent.getActivity(
            this, 0, Intent(this, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE
        )
        val stop = PendingIntent.getService(
            this, 1,
            Intent(this, BubbleOverlayService::class.java).setAction(ACTION_STOP),
            PendingIntent.FLAG_IMMUTABLE
        )
        val notification = NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle(getString(R.string.notif_title))
            .setContentText(getString(R.string.notif_text))
            .setSmallIcon(R.drawable.ic_bubble)
            .setContentIntent(open)
            .setOngoing(true)
            .addAction(R.drawable.ic_stop, getString(R.string.notif_stop), stop)
            .build()
        startForeground(NOTIFICATION_ID, notification)
    }

    // ---- rendering (main thread only)

    private fun overlayParams(width: Int, height: Int, x: Int, y: Int) = WindowManager.LayoutParams(
        width, height,
        WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
        WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
        PixelFormat.TRANSLUCENT
    ).apply {
        gravity = Gravity.TOP or Gravity.START
        this.x = x
        this.y = y
    }

    /** Updates views in place so a dragged bubble keeps its position when state changes. */
    private fun render() {
        val current = manager.snapshot()
        val live = current.map { it.pkg }.toSet()
        handles.keys.filter { it !in live }.forEach { removeHandle(it) }

        current.forEachIndexed { index, bubble ->
            val handle = handles[bubble.pkg]
                ?: createBubble(bubble.pkg, index)?.also { handles[bubble.pkg] = it }
                ?: return@forEachIndexed
            handle.icon.setBackgroundResource(
                if (bubble.expanded) R.drawable.bg_bubble_active else R.drawable.bg_bubble
            )
        }
        updateChip(manager.recentsSnapshot().isNotEmpty(), current.size)
    }

    private fun createBubble(pkg: String, index: Int): BubbleHandle? {
        val root = LayoutInflater.from(themed).inflate(R.layout.view_bubble, null)
        val icon = root.findViewById<ImageView>(R.id.bubble_icon)
        icon.setImageDrawable(runCatching { packageManager.getApplicationIcon(pkg) }.getOrNull())
        root.contentDescription = getString(R.string.cd_bubble, labelOf(pkg))

        val size = dp(BUBBLE_DP)
        val screen = resources.displayMetrics
        val position = positions.getOrPut(pkg) {
            Point(screen.widthPixels - size, dp(FIRST_SLOT_TOP_DP) + index * dp(SLOT_STEP_DP))
        }
        val params = overlayParams(size, size, position.x, position.y)
        attachDrag(root, params, pkg)
        return try {
            wm.addView(root, params)
            BubbleHandle(root, icon, params)
        } catch (e: Exception) {
            DiagLog.log(TAG, "Could not show bubble for $pkg: ${e.message}")
            null
        }
    }

    private fun removeHandle(pkg: String) {
        handles.remove(pkg)?.let { runCatching { wm.removeView(it.root) } }
        positions.remove(pkg)
    }

    private fun labelOf(pkg: String): String = runCatching {
        packageManager.getApplicationLabel(packageManager.getApplicationInfo(pkg, 0)).toString()
    }.getOrDefault(pkg)

    // ---- drag, tap and drop-to-close

    private fun attachDrag(view: View, params: WindowManager.LayoutParams, pkg: String) {
        val slop = ViewConfiguration.get(this).scaledTouchSlop
        var startX = 0
        var startY = 0
        var downX = 0f
        var downY = 0f
        var dragging = false

        view.setOnTouchListener { _, event ->
            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    startX = params.x
                    startY = params.y
                    downX = event.rawX
                    downY = event.rawY
                    dragging = false
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val dx = event.rawX - downX
                    val dy = event.rawY - downY
                    if (!dragging && hypot(dx, dy) > slop) {
                        dragging = true
                        showCloseZone()
                    }
                    if (dragging) {
                        params.x = startX + dx.toInt()
                        params.y = startY + dy.toInt()
                        wm.updateViewLayout(view, params)
                        setCloseHot(isOverClose(event.rawX, event.rawY))
                    }
                    true
                }
                MotionEvent.ACTION_UP -> {
                    if (!dragging) {
                        worker.execute { manager.toggle(pkg) }
                    } else {
                        val dropped = isOverClose(event.rawX, event.rawY)
                        hideCloseZone()
                        if (dropped) {
                            worker.execute { manager.dismiss(pkg) }
                        } else {
                            snapToEdge(view, params)
                            positions[pkg] = Point(params.x, params.y)
                        }
                    }
                    true
                }
                MotionEvent.ACTION_CANCEL -> {
                    hideCloseZone()
                    true
                }
                else -> false
            }
        }
    }

    private fun snapToEdge(view: View, params: WindowManager.LayoutParams) {
        val width = resources.displayMetrics.widthPixels
        params.x = if (params.x + params.width / 2 < width / 2) 0 else width - params.width
        wm.updateViewLayout(view, params)
    }

    private fun closeZoneTop(): Int =
        resources.displayMetrics.heightPixels - dp(CLOSE_ZONE_DP) - dp(CLOSE_ZONE_BOTTOM_DP)

    private fun isOverClose(rawX: Float, rawY: Float): Boolean {
        val centerX = resources.displayMetrics.widthPixels / 2f
        val centerY = closeZoneTop() + dp(CLOSE_ZONE_DP) / 2f
        return hypot(rawX - centerX, rawY - centerY) < dp(CLOSE_HIT_RADIUS_DP)
    }

    private fun showCloseZone() {
        if (closeZone != null) return
        val view = LayoutInflater.from(themed).inflate(R.layout.view_close_zone, null)
        val size = dp(CLOSE_ZONE_DP)
        val x = (resources.displayMetrics.widthPixels - size) / 2
        runCatching { wm.addView(view, overlayParams(size, size, x, closeZoneTop())) }
            .onSuccess { closeZone = view }
    }

    private fun setCloseHot(hot: Boolean) {
        val zone = closeZone ?: return
        zone.findViewById<View>(R.id.close_bg).setBackgroundResource(
            if (hot) R.drawable.bg_close_zone_hot else R.drawable.bg_close_zone
        )
        val iconAttr = if (hot) {
            com.google.android.material.R.attr.colorOnError
        } else {
            com.google.android.material.R.attr.colorOnErrorContainer
        }
        zone.findViewById<ImageView>(R.id.close_icon).imageTintList =
            ColorStateList.valueOf(MaterialColors.getColor(themed, iconAttr, 0))
        val scale = if (hot) 1.2f else 1f
        zone.animate().scaleX(scale).scaleY(scale).setDuration(100).start()
    }

    private fun hideCloseZone() {
        closeZone?.let { runCatching { wm.removeView(it) } }
        closeZone = null
    }

    // ---- recents chip and panel

    private fun updateChip(visible: Boolean, bubbleCount: Int) {
        if (!visible) {
            hideRecents()
            chip?.let { runCatching { wm.removeView(it.root) } }
            chip = null
            return
        }
        val size = dp(BUBBLE_DP)
        val x = resources.displayMetrics.widthPixels - size
        val y = dp(FIRST_SLOT_TOP_DP) + bubbleCount * dp(SLOT_STEP_DP)

        chip?.let {
            it.params.x = x
            it.params.y = y
            wm.updateViewLayout(it.root, it.params)
            return
        }
        val root = LayoutInflater.from(themed).inflate(R.layout.view_bubble, null)
        val icon = root.findViewById<ImageView>(R.id.bubble_icon)
        icon.setImageResource(R.drawable.ic_add)
        icon.setBackgroundResource(R.drawable.bg_chip)
        icon.imageTintList = ColorStateList.valueOf(
            MaterialColors.getColor(themed, com.google.android.material.R.attr.colorOnSecondaryContainer, 0)
        )
        root.contentDescription = getString(R.string.cd_recents)
        root.setOnClickListener { if (recentsPanel == null) showRecents() else hideRecents() }

        val params = overlayParams(size, size, x, y)
        runCatching { wm.addView(root, params) }.onSuccess { chip = BubbleHandle(root, icon, params) }
    }

    private fun showRecents() {
        val recents = manager.recentsSnapshot()
        val anchor = chip ?: return
        if (recents.isEmpty()) return

        val inflater = LayoutInflater.from(themed)
        val panel = inflater.inflate(R.layout.view_recents_panel, null)
        val list = panel.findViewById<LinearLayout>(R.id.recents_list)
        recents.forEach { pkg ->
            val row = inflater.inflate(R.layout.item_app, list, false)
            row.findViewById<ImageView>(R.id.app_icon)
                .setImageDrawable(runCatching { packageManager.getApplicationIcon(pkg) }.getOrNull())
            row.findViewById<TextView>(R.id.app_label).text = labelOf(pkg)
            row.findViewById<TextView>(R.id.app_package).visibility = View.GONE
            row.setOnClickListener {
                hideRecents()
                worker.execute { manager.add(pkg) }
            }
            list.addView(row)
        }
        panel.setOnTouchListener { _, event ->
            if (event.action == MotionEvent.ACTION_OUTSIDE) {
                hideRecents()
                true
            } else {
                false
            }
        }
        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_WATCH_OUTSIDE_TOUCH,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.END
            x = dp(BUBBLE_DP + 8)
            y = anchor.params.y
        }
        runCatching { wm.addView(panel, params) }.onSuccess { recentsPanel = panel }
    }

    private fun hideRecents() {
        recentsPanel?.let { runCatching { wm.removeView(it) } }
        recentsPanel = null
    }
}

package com.multitask.bubble.ui

import android.content.Intent
import android.graphics.drawable.Drawable
import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.appcompat.app.AppCompatActivity
import androidx.core.widget.doAfterTextChanged
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.google.android.material.bottomsheet.BottomSheetBehavior
import com.google.android.material.bottomsheet.BottomSheetDialog
import com.multitask.bubble.databinding.ItemAppBinding
import com.multitask.bubble.databinding.SheetAppPickerBinding
import kotlin.concurrent.thread

/** Bottom sheet listing launchable apps with a search field. */
class AppPickerSheet(
    private val activity: AppCompatActivity,
    private val onPick: (packageName: String) -> Unit
) {
    private class Entry(val packageName: String, val label: String, val icon: Drawable)

    fun show() {
        val dialog = BottomSheetDialog(activity)
        val binding = SheetAppPickerBinding.inflate(LayoutInflater.from(activity))
        val adapter = Adapter { entry ->
            dialog.dismiss()
            onPick(entry.packageName)
        }
        binding.list.layoutManager = LinearLayoutManager(activity)
        binding.list.adapter = adapter
        binding.search.doAfterTextChanged { adapter.filter(it?.toString().orEmpty()) }

        dialog.setContentView(binding.root)
        dialog.behavior.skipCollapsed = true
        dialog.behavior.state = BottomSheetBehavior.STATE_EXPANDED
        dialog.show()

        thread {
            val apps = loadApps()
            activity.runOnUiThread { adapter.submit(apps) }
        }
    }

    private fun loadApps(): List<Entry> {
        val pm = activity.packageManager
        val launcher = Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER)
        return pm.queryIntentActivities(launcher, 0)
            .filter { it.activityInfo.packageName != activity.packageName }
            .distinctBy { it.activityInfo.packageName }
            .map { Entry(it.activityInfo.packageName, it.loadLabel(pm).toString(), it.loadIcon(pm)) }
            .sortedBy { it.label.lowercase() }
    }

    private class Adapter(
        private val onClick: (Entry) -> Unit
    ) : RecyclerView.Adapter<Adapter.Holder>() {

        private var all: List<Entry> = emptyList()
        private var shown: List<Entry> = emptyList()
        private var query = ""

        class Holder(val binding: ItemAppBinding) : RecyclerView.ViewHolder(binding.root)

        fun submit(items: List<Entry>) {
            all = items
            filter(query)
        }

        fun filter(text: String) {
            query = text
            shown = if (text.isBlank()) all else all.filter { it.label.contains(text, ignoreCase = true) }
            notifyDataSetChanged()
        }

        override fun onCreateViewHolder(parent: ViewGroup, viewType: Int) =
            Holder(ItemAppBinding.inflate(LayoutInflater.from(parent.context), parent, false))

        override fun getItemCount() = shown.size

        override fun onBindViewHolder(holder: Holder, position: Int) {
            val entry = shown[position]
            holder.binding.appIcon.setImageDrawable(entry.icon)
            holder.binding.appLabel.text = entry.label
            holder.binding.appPackage.text = entry.packageName
            holder.binding.root.setOnClickListener { onClick(entry) }
        }
    }
}

===== FILE: app/src/test/java/com/multitask/bubble/BubbleManagerTest.kt =====
package com.multitask.bubble

import android.content.Context
import android.graphics.Rect
import androidx.test.core.app.ApplicationProvider
import com.multitask.bubble.core.BubbleManager
import com.multitask.bubble.core.WindowBackend
import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Test
import org.junit.runner.RunWith
import org.robolectric.RobolectricTestRunner
import org.robolectric.annotation.Config

@RunWith(RobolectricTestRunner::class)
@Config(sdk = [34])
class BubbleManagerTest {

    private class FakeWindowBackend : WindowBackend {
        val launched = mutableListOf<String>()
        val closed = mutableListOf<String>()
        val moves = mutableListOf<Pair<Int, Rect>>()

        override fun isReady() = true
        override fun requestPermission(onResult: (Boolean) -> Unit) = onResult(true)
        override fun connect(onConnected: () -> Unit) = onConnected()
        override fun disconnect() {}
        override fun setupFreeform() {}
        override fun launch(packageName: String, bounds: Rect): Int {
            launched.add(packageName)
            return launched.size
        }
        override fun moveTask(taskId: Int, bounds: Rect) {
            moves.add(taskId to bounds)
        }
        override fun close(packageName: String) {
            closed.add(packageName)
        }
    }

    private val context: Context get() = ApplicationProvider.getApplicationContext()

    @Test
    fun capEvictsTheOldestBubble() {
        val backend = FakeWindowBackend()
        val manager = BubbleManager(context, backend) {}

        manager.add("app.one")
        manager.add("app.two")
        manager.add("app.three")
        assertEquals(BubbleManager.MAX_BUBBLES, manager.snapshot().size)

        manager.add("app.four")

        val pkgs = manager.snapshot().map { it.pkg }
        assertEquals(BubbleManager.MAX_BUBBLES, pkgs.size)
        assertFalse(pkgs.contains("app.one"))
        assertEquals("app.one", manager.recentsSnapshot().first())
        assertTrue(backend.closed.contains("app.one"))
    }

    @Test
    fun onlyOneBubbleIsOpenAtATime() {
        val backend = FakeWindowBackend()
        val manager = BubbleManager(context, backend) {}

        manager.add("app.one")
        manager.add("app.two")

        val state = manager.snapshot().associateBy { it.pkg }
        assertFalse(state.getValue("app.one").expanded)
        assertTrue(state.getValue("app.two").expanded)
        assertTrue(backend.moves.any { it.first == 1 }) // app.one was moved out of the way
    }

    @Test
    fun addingAnExistingAppDoesNotDuplicateIt() {
        val manager = BubbleManager(context, FakeWindowBackend()) {}

        manager.add("app.one")
        manager.add("app.one")

        assertEquals(1, manager.snapshot().size)
    }

    @Test
    fun dismissedBubblesMoveToRecentsAndReopen() {
        val backend = FakeWindowBackend()
        val manager = BubbleManager(context, backend) {}

        manager.add("app.one")
        manager.dismiss("app.one")
        assertTrue(manager.snapshot().isEmpty())
        assertEquals(listOf("app.one"), manager.recentsSnapshot())

        manager.add("app.one")
        assertEquals(1, manager.snapshot().size)
        assertTrue(manager.recentsSnapshot().isEmpty())
    }

    @Test
    fun bubblesAreRestoredCollapsedByANewManager() {
        val first = BubbleManager(context, FakeWindowBackend()) {}
        first.add("app.one")

        val restored = BubbleManager(context, FakeWindowBackend()) {}.snapshot()

        assertEquals(listOf("app.one"), restored.map { it.pkg })
        assertFalse(restored.first().expanded)
    }
}
