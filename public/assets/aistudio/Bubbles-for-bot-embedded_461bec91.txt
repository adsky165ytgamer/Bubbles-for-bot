HOW IT WORKS (Android 17 style bubbles at the framework level)
- Each bubble owns a private VirtualDisplay (public DisplayManager API). Its pixels are drawn into a SurfaceView inside our own overlay window, so the app really lives inside our window and we own the frame, minimize and close.
- Shizuku (shell UID) starts the app on that display with "am start --display <id>" and injects touch/back events with the hidden input service. No OEM code, no split screen, no system freeform window.
- Minimize hides our window and detaches the surface; the app keeps running. Close force-stops the app and releases the display.
- Fallback: if the ROM lacks the secondary-display feature, or launching on our display fails, bubbles switch to the system freeform window (am start --windowingMode 5, placed with am task resize). This is best effort and shows the OEM frame.
- Bubble state (cap of 3, recents, persistence) is pure logic in BubbleManager and is unit tested.

KNOWN GAPS
- Keyboard typing inside embedded apps is not supported yet (taps, scrolling, swipes and the back button work). Single finger only, no pinch.
- Window size is fixed at 80% x 60% of the screen.
- Each embedded app costs RAM and GPU. If the phone lags, lower MAX_BUBBLES in BubbleManager.kt.
- Resizing and per-bubble window position memory are not implemented.

package com.multitask.bubble.core

import android.content.Context

/**
 * Pure bubble state: the list, the cap, recents and persistence. Main thread only, never blocks.
 * The caller reacts to the returned [Change] (hide or release windows).
 */
class BubbleManager(
    context: Context,
    private val onChanged: () -> Unit
) {
    data class Bubble(val pkg: String, val expanded: Boolean = false)

    /** [evicted] bubbles were removed to make room. [collapsed] bubbles were open and got hidden. */
    data class Change(
        val evicted: List<String> = emptyList(),
        val collapsed: List<String> = emptyList()
    )

    companion object {
        const val MAX_BUBBLES = 3
        const val MAX_RECENTS = 8
        private const val PREFS = "bubbles"
        private const val KEY_ACTIVE = "active"
        private const val KEY_RECENTS = "recents"
    }

    private val prefs = context.applicationContext.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
    private val bubbles = mutableListOf<Bubble>()
    private val recents = mutableListOf<String>()

    init {
        restore()
    }

    fun snapshot(): List<Bubble> = bubbles.toList()

    fun recentsSnapshot(): List<String> = recents.toList()

    /** Adds the app if needed (evicting the oldest at the cap) and makes it the one open bubble. */
    fun open(pkg: String): Change {
        val evicted = mutableListOf<String>()
        if (bubbles.none { it.pkg == pkg }) {
            if (bubbles.size >= MAX_BUBBLES) {
                val oldest = bubbles.first().pkg
                remove(oldest)
                evicted.add(oldest)
            }
            bubbles.add(Bubble(pkg))
            recents.remove(pkg)
        }
        val collapsed = bubbles.filter { it.expanded && it.pkg != pkg }.map { it.pkg }
        bubbles.replaceAll { it.copy(expanded = it.pkg == pkg) }
        commit()
        return Change(evicted, collapsed)
    }

    fun collapse(pkg: String) {
        bubbles.replaceAll { if (it.pkg == pkg) it.copy(expanded = false) else it }
        commit()
    }

    /** Removes the bubble and remembers it in recents. */
    fun dismiss(pkg: String) {
        if (bubbles.none { it.pkg == pkg }) return
        remove(pkg)
        commit()
    }

    private fun remove(pkg: String) {
        bubbles.removeAll { it.pkg == pkg }
        recents.remove(pkg)
        recents.add(0, pkg)
        while (recents.size > MAX_RECENTS) recents.removeAt(recents.lastIndex)
    }

    private fun commit() {
        prefs.edit()
            .putString(KEY_ACTIVE, bubbles.joinToString(",") { it.pkg })
            .putString(KEY_RECENTS, recents.joinToString(","))
            .apply()
        onChanged()
    }

    /** Restored bubbles start collapsed; their apps start fresh on first tap. */
    private fun restore() {
        prefs.getString(KEY_ACTIVE, "").orEmpty().split(",").filter { it.isNotBlank() }
            .take(MAX_BUBBLES).forEach { bubbles.add(Bubble(it)) }
        prefs.getString(KEY_RECENTS, "").orEmpty().split(",").filter { it.isNotBlank() }
            .take(MAX_RECENTS).forEach { recents.add(it) }
    }
}

package com.multitask.bubble.core

import android.util.Log
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

/** In-memory log shared by the activity and the overlay service, mirrored to logcat. */
object DiagLog {
    private const val MAX_CHARS = 6000
    private const val TRIM_CHARS = 2000

    private val buffer = StringBuilder()

    /** Called on whichever thread logged. UI code must post to the main thread itself. */
    @Volatile
    var listener: ((String) -> Unit)? = null

    fun log(tag: String, message: String) {
        Log.d(tag, message)
        val stamp = SimpleDateFormat("HH:mm:ss", Locale.getDefault()).format(Date())
        synchronized(buffer) {
            buffer.append('[').append(stamp).append("] ").append(message).append('\n')
            if (buffer.length > MAX_CHARS) buffer.delete(0, TRIM_CHARS)
        }
        listener?.invoke(snapshot())
    }

    fun snapshot(): String = synchronized(buffer) { buffer.toString() }

    fun clear() {
        synchronized(buffer) { buffer.setLength(0) }
        listener?.invoke("")
    }
}

import android.os.IBinder
import android.os.SystemClock
import android.view.InputDevice
import android.view.InputEvent
import android.view.KeyEvent
import android.view.MotionEvent
import kotlin.system.exitProcess

/**
 * Runs inside the Shizuku shell process (UID 2000). Hidden APIs are reachable here because this
 * is not a normal app process.
 */
class ShellService : IShellService.Stub() {

    private val inputManager: Any? by lazy {
        runCatching {
            val binder = Class.forName("android.os.ServiceManager")
                .getMethod("getService", String::class.java)
                .invoke(null, "input") as IBinder
            Class.forName("android.hardware.input.IInputManager\$Stub")
                .getMethod("asInterface", IBinder::class.java)
                .invoke(null, binder)
        }.getOrNull()
    }

    private val injectMethod by lazy {
        runCatching {
            Class.forName("android.hardware.input.IInputManager")
                .getMethod("injectInputEvent", InputEvent::class.java, Int::class.javaPrimitiveType)
        }.getOrNull()
    }

    private val setDisplayId by lazy {
        runCatching {
            InputEvent::class.java.getMethod("setDisplayId", Int::class.javaPrimitiveType)
        }.getOrNull()
    }

    override fun destroy() {
        exitProcess(0)
    }

    override fun exec(command: String): String {
        return try {
            val process = ProcessBuilder("sh", "-c", command)
                .redirectErrorStream(true)
                .start()
            val output = process.inputStream.bufferedReader().readText()
            process.waitFor()
            output
        } catch (e: Exception) {
            "[exception] ${e.message}"
        }
    }

    override fun injectMotion(displayId: Int, action: Int, x: Float, y: Float, downTime: Long): Boolean {
        val event = MotionEvent.obtain(downTime, SystemClock.uptimeMillis(), action, x, y, 0)
        event.source = InputDevice.SOURCE_TOUCHSCREEN
        return try {
            inject(event, displayId)
        } finally {
            event.recycle()
        }
    }

    override fun injectKey(displayId: Int, keyCode: Int): Boolean {
        val now = SystemClock.uptimeMillis()
        val down = KeyEvent(now, now, KeyEvent.ACTION_DOWN, keyCode, 0)
        val up = KeyEvent(now, now, KeyEvent.ACTION_UP, keyCode, 0)
        down.source = InputDevice.SOURCE_KEYBOARD
        up.source = InputDevice.SOURCE_KEYBOARD
        return inject(down, displayId) && inject(up, displayId)
    }

    private fun inject(event: InputEvent, displayId: Int): Boolean = try {
        setDisplayId?.invoke(event, displayId)
        // 0 = INJECT_INPUT_EVENT_MODE_ASYNC
        injectMethod?.invoke(inputManager, event, 0) as? Boolean ?: false
    } catch (e: Exception) {
        false
    }
}

    }

    override fun isReady(): Boolean = try {
        Shizuku.pingBinder() && Shizuku.checkSelfPermission() == PackageManager.PERMISSION_GRANTED
    } catch (e: Exception) {
        false
    }

    override fun requestPermission(onResult: (Boolean) -> Unit) {
        if (!Shizuku.pingBinder()) {
            onResult(false)
            return
        }
        if (Shizuku.checkSelfPermission() == PackageManager.PERMISSION_GRANTED) {
            onResult(true)
            return
        }
        val listener = object : Shizuku.OnRequestPermissionResultListener {
            override fun onRequestPermissionResult(requestCode: Int, grantResult: Int) {
                if (requestCode != REQUEST_CODE) return
                Shizuku.removeRequestPermissionResultListener(this)
                val granted = grantResult == PackageManager.PERMISSION_GRANTED
                DiagLog.log(TAG, "Shizuku permission ${if (granted) "granted" else "denied"}")
                onResult(granted)
            }
        }
        Shizuku.addRequestPermissionResultListener(listener)
        Shizuku.requestPermission(REQUEST_CODE)
    }

    override fun connect(onConnected: () -> Unit) {
        if (shell != null) {
            onConnected()
            return
        }
        if (!isReady()) {
            DiagLog.log(TAG, "Cannot connect: Shizuku is not running or not authorized")
            return
        }
        val firstWaiter = synchronized(waiting) {
            waiting.add(onConnected)
            waiting.size == 1
        }
        if (firstWaiter) {
            DiagLog.log(TAG, "Binding shell service")
            Shizuku.bindUserService(userServiceArgs, connection)
        }
    }

    override fun disconnect() {
        if (shell == null) return
        runCatching { Shizuku.unbindUserService(userServiceArgs, connection, true) }
        shell = null
        prepared = false
    }

    override fun prepare() {
        if (prepared) return
        // Lets apps that are not marked resizable fill a secondary display or a freeform window.
        sh("settings put global force_resizable_activities 1")
        sh("settings put global enable_freeform_support 1")
        prepared = true
    }

    override fun launchOnDisplay(packageName: String, displayId: Int): Boolean {
        val component = context.packageManager.getLaunchIntentForPackage(packageName)
            ?.component?.flattenToShortString()
        if (component == null) {
            DiagLog.log(TAG, "No launcher activity found for $packageName")
            return false
        }
        val output = sh("am start -n $component --display $displayId")
        val failed = output.contains("Error", ignoreCase = true) ||
            output.contains("Exception", ignoreCase = true)
        if (failed) DiagLog.log(TAG, "Launch on display $displayId failed for $packageName")
        return !failed
    }

    override fun launchFreeform(packageName: String, bounds: Rect): Int {
        val component = context.packageManager.getLaunchIntentForPackage(packageName)
            ?.component?.flattenToShortString()
        if (component == null) {
            DiagLog.log(TAG, "No launcher activity found for $packageName")
            return -1
        }
        sh("am start -n $component --windowingMode $WINDOWING_MODE_FREEFORM")
        var taskId = -1
        for (attempt in 0 until TASK_POLL_ATTEMPTS) {
            taskId = findTaskId(packageName)
            if (taskId > 0) break
            Thread.sleep(TASK_POLL_DELAY_MS)
        }
        if (taskId > 0) moveTask(taskId, bounds)
        else DiagLog.log(TAG, "Could not find a task for $packageName, so it cannot be moved")
        return taskId
    }

    override fun moveTask(taskId: Int, bounds: Rect) {
        sh("am task resize $taskId ${bounds.left},${bounds.top},${bounds.right},${bounds.bottom}")
    }

    override fun injectMotion(displayId: Int, action: Int, x: Float, y: Float, downTime: Long) {
        val ok = try {
            shell?.injectMotion(displayId, action, x, y, downTime) ?: false
        } catch (e: Exception) {
            false
        }
        if (!ok) logInjectFailureOnce()
    }

    override fun injectKey(displayId: Int, keyCode: Int) {
        val ok = try {
            shell?.injectKey(displayId, keyCode) ?: false
        } catch (e: Exception) {
            false
        }
        if (!ok) logInjectFailureOnce()
    }

    override fun close(packageName: String) {
        sh("am force-stop $packageName")
    }

    private fun findTaskId(packageName: String): Int {
        val dump = sh("dumpsys activity recents", quiet = true)
        val pattern = Regex("""Task\{\S+ #(\d+)[^}]*?A=(?:\d+:)?${Regex.escape(packageName)}\b""")
        return pattern.find(dump)?.groupValues?.get(1)?.toIntOrNull() ?: -1
    }

    private fun logInjectFailureOnce() {
        if (injectFailureLogged) return
        injectFailureLogged = true
        DiagLog.log(TAG, "Input injection failed. Touch and keys will not reach the embedded app")
    }

    private fun sh(command: String, quiet: Boolean = false): String {
        val service = shell
        if (service == null) {
            DiagLog.log(TAG, "$command -> no shell connected")
            return ""
        }
        val output = try {
            service.exec(command)
        } catch (e: Exception) {
            "[remote error] ${e.message}"
        }
        if (!quiet) {
            val trimmed = output.trim().take(LOG_OUTPUT_LIMIT)
            DiagLog.log(TAG, if (trimmed.isEmpty()) command else "$command -> $trimmed")
        }
        return output
    }
}

    private lateinit var wm: WindowManager
    private lateinit var backend: ShizukuWindowBackend
    private lateinit var manager: BubbleManager

    private val ui = Handler(Looper.getMainLooper())
    private val worker = Executors.newSingleThreadExecutor()
    private val handles = mutableMapOf<String, BubbleHandle>()
    private val positions = mutableMapOf<String, Point>()
    private val windows = mutableMapOf<String, BubbleWindow>()
    private val pendingAdds = mutableListOf<String>()
    private var ready = false

    // Embedded bubbles need secondary-display support. If the ROM lacks it, or a launch on our
    // virtual display fails, every bubble falls back to the system's freeform windows.
    private var embeddedBroken = false
    private val embeddedSupported by lazy { packageManager.hasSystemFeature(FEATURE_SECONDARY_DISPLAYS) }

    private var chip: BubbleHandle? = null
    private var closeZone: View? = null
    private var recentsPanel: View? = null

    private val density by lazy { resources.displayMetrics.density }
    private fun dp(value: Int) = (value * density).toInt()

    override fun onCreate() {
        super.onCreate()
        running = true
        themed = ContextThemeWrapper(this, R.style.Theme_BubbleManager)
        wm = getSystemService(WINDOW_SERVICE) as WindowManager
        backend = ShizukuWindowBackend(applicationContext)
        manager = BubbleManager(applicationContext) { render() }

        startAsForeground()
        render() // bubbles restored from the last session show up right away

        backend.connect {
            worker.execute {
                backend.prepare()
                ui.post {
                    ready = true
                    DiagLog.log(TAG, if (embeddedSupported) "Mode: embedded bubbles" else "Mode: freeform fallback (no secondary display support)")
                    pendingAdds.toList().also { pendingAdds.clear() }.forEach { openBubble(it) }
                }
            }
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> stopSelf()
            ACTION_ADD -> intent.getStringExtra(EXTRA_PKG)?.let { openBubble(it) }
        }
        return START_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onDestroy() {
        running = false
        hideRecents()
        hideCloseZone()
        windows.values.forEach { it.release() }
        windows.clear()
        handles.keys.toList().forEach { removeHandle(it) }
        chip?.let { runCatching { wm.removeView(it.root) } }
        chip = null
        worker.execute { backend.disconnect() } // queued after the force-stops above

    private fun createWindow(pkg: String): BubbleWindow =

            EmbeddedWindow(
                context = themed, wm = wm, backend = backend, worker = worker, pkg = pkg,
                onMinimize = { minimizeBubble(pkg) },
                onClose = { closeBubble(pkg) },
                onFailed = { onEmbeddedFailed(pkg) }
            )
        } else {
            FreeformWindow(this, backend, worker, pkg)
        }

    private fun openBubble(pkg: String) {
        if (!ready) {
            if (pkg !in pendingAdds) pendingAdds.add(pkg)
            return
        }
        val change = manager.open(pkg)
        change.evicted.forEach { windows.remove(it)?.release() }
        change.collapsed.forEach { windows[it]?.hide() }
        windows.getOrPut(pkg) { createWindow(pkg) }.show()
    }

    private fun minimizeBubble(pkg: String) {
        manager.collapse(pkg)
        windows[pkg]?.hide()
    }

    private fun closeBubble(pkg: String) {
        manager.dismiss(pkg)
        windows.remove(pkg)?.release()
    }

    private fun onBubbleTap(pkg: String) {
        val open = manager.snapshot().firstOrNull { it.pkg == pkg }?.expanded == true
        if (open) minimizeBubble(pkg) else openBubble(pkg)
    }

    private fun onEmbeddedFailed(pkg: String) {
        embeddedBroken = true
        windows.remove(pkg)?.release()
        ui.postDelayed({
            if (manager.snapshot().any { it.pkg == pkg }) openBubble(pkg)
        }, 400)
    }

    // ---- rendering (main thread only)

    private fun overlayParams(width: Int, height: Int, x: Int, y: Int) = WindowManager.LayoutParams(
        width, height,
        WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
        WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
        PixelFormat.TRANSLUCENT
    ).apply {
        gravity = Gravity.TOP or Gravity.START
        this.x = x
        this.y = y
    }

    /** Updates views in place so a dragged bubble keeps its position when state changes. */
    private fun render() {
        val current = manager.snapshot()
        val live = current.map { it.pkg }.toSet()
        handles.keys.filter { it !in live }.forEach { removeHandle(it) }

        current.forEachIndexed { index, bubble ->
            val handle = handles[bubble.pkg]
                ?: createBubble(bubble.pkg, index)?.also { handles[bubble.pkg] = it }
                ?: return@forEachIndexed
            handle.icon.setBackgroundResource(
                if (bubble.expanded) R.drawable.bg_bubble_active else R.drawable.bg_bubble
            )
        }
        updateChip(manager.recentsSnapshot().isNotEmpty(), current.size)
    }

    private fun createBubble(pkg: String, index: Int): BubbleHandle? {
        val root = LayoutInflater.from(themed).inflate(R.layout.view_bubble, null)
        val icon = root.findViewById<ImageView>(R.id.bubble_icon)
        icon.setImageDrawable(runCatching { packageManager.getApplicationIcon(pkg) }.getOrNull())
        root.contentDescription = getString(R.string.cd_bubble, labelOf(pkg))

        val size = dp(BUBBLE_DP)
        val screen = resources.displayMetrics
        val position = positions.getOrPut(pkg) {
            Point(screen.widthPixels - size, dp(FIRST_SLOT_TOP_DP) + index * dp(SLOT_STEP_DP))
        }
        val params = overlayParams(size, size, position.x, position.y)
        attachDrag(root, params, pkg)
        return try {
            wm.addView(root, params)
            BubbleHandle(root, icon, params)
        } catch (e: Exception) {
            DiagLog.log(TAG, "Could not show bubble for $pkg: ${e.message}")
            null
        }
    }

    private fun removeHandle(pkg: String) {
        handles.remove(pkg)?.let { runCatching { wm.removeView(it.root) } }
        positions.remove(pkg)
    }

    private fun labelOf(pkg: String): String = runCatching {
        packageManager.getApplicationLabel(packageManager.getApplicationInfo(pkg, 0)).toString()
    }.getOrDefault(pkg)

    // ---- drag, tap and drop-to-close

    private fun attachDrag(view: View, params: WindowManager.LayoutParams, pkg: String) {
        val slop = ViewConfiguration.get(this).scaledTouchSlop
        var startX = 0
        var startY = 0
        var downX = 0f
        var downY = 0f
        var dragging = false

        view.setOnTouchListener { _, event ->
            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    startX = params.x
                    startY = params.y
                    downX = event.rawX
                    downY = event.rawY
                    dragging = false
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val dx = event.rawX - downX
                    val dy = event.rawY - downY
                    if (!dragging && hypot(dx, dy) > slop) {
                        dragging = true
                        showCloseZone()
                    }
                    if (dragging) {
                        params.x = startX + dx.toInt()
                        params.y = startY + dy.toInt()
                        wm.updateViewLayout(view, params)
                        setCloseHot(isOverClose(event.rawX, event.rawY))
                    }
                    true
                }
                MotionEvent.ACTION_UP -> {
                    if (!dragging) {
                        onBubbleTap(pkg)
                    } else {
                        val dropped = isOverClose(event.rawX, event.rawY)
                        hideCloseZone()
                        if (dropped) {
                            closeBubble(pkg)
                        } else {
                            snapToEdge(view, params)
                            positions[pkg] = Point(params.x, params.y)
                        }
                    }
                    true
                }
                MotionEvent.ACTION_CANCEL -> {
                    hideCloseZone()
                    true
                }
                else -> false
            }
        }
    }

    private fun snapToEdge(view: View, params: WindowManager.LayoutParams) {
        val width = resources.displayMetrics.widthPixels
        params.x = if (params.x + params.width / 2 < width / 2) 0 else width - params.width
        wm.updateViewLayout(view, params)
    }

    private fun closeZoneTop(): Int =
        resources.displayMetrics.heightPixels - dp(CLOSE_ZONE_DP) - dp(CLOSE_ZONE_BOTTOM_DP)

    private fun isOverClose(rawX: Float, rawY: Float): Boolean {
        val centerX = resources.displayMetrics.widthPixels / 2f
        val centerY = closeZoneTop() + dp(CLOSE_ZONE_DP) / 2f
        return hypot(rawX - centerX, rawY - centerY) < dp(CLOSE_HIT_RADIUS_DP)
    }

    private fun showCloseZone() {
        if (closeZone != null) return
        val view = LayoutInflater.from(themed).inflate(R.layout.view_close_zone, null)
        val size = dp(CLOSE_ZONE_DP)
        val x = (resources.displayMetrics.widthPixels - size) / 2
        runCatching { wm.addView(view, overlayParams(size, size, x, closeZoneTop())) }
            .onSuccess { closeZone = view }
    }

    private fun setCloseHot(hot: Boolean) {
        val zone = closeZone ?: return
        zone.findViewById<View>(R.id.close_bg).setBackgroundResource(
            if (hot) R.drawable.bg_close_zone_hot else R.drawable.bg_close_zone
        )
        val iconAttr = if (hot) {
            com.google.android.material.R.attr.colorOnError
        } else {
            com.google.android.material.R.attr.colorOnErrorContainer
        }
        zone.findViewById<ImageView>(R.id.close_icon).imageTintList =
            ColorStateList.valueOf(MaterialColors.getColor(themed, iconAttr, 0))
        val scale = if (hot) 1.2f else 1f
        zone.animate().scaleX(scale).scaleY(scale).setDuration(100).start()
    }

    private fun hideCloseZone() {
        closeZone?.let { runCatching { wm.removeView(it) } }
        closeZone = null
    }

    // ---- recents chip and panel

    private fun updateChip(visible: Boolean, bubbleCount: Int) {
        if (!visible) {
            hideRecents()
            chip?.let { runCatching { wm.removeView(it.root) } }
            chip = null
            return
        }
        val size = dp(BUBBLE_DP)
        val x = resources.displayMetrics.widthPixels - size
        val y = dp(FIRST_SLOT_TOP_DP) + bubbleCount * dp(SLOT_STEP_DP)

        chip?.let {
            it.params.x = x
            it.params.y = y
            wm.updateViewLayout(it.root, it.params)
            return
        }
        val root = LayoutInflater.from(themed).inflate(R.layout.view_bubble, null)
        val icon = root.findViewById<ImageView>(R.id.bubble_icon)
        icon.setImageResource(R.drawable.ic_add)
        icon.setBackgroundResource(R.drawable.bg_chip)
        icon.imageTintList = ColorStateList.valueOf(
            MaterialColors.getColor(themed, com.google.android.material.R.attr.colorOnSecondaryContainer, 0)
        )
        root.contentDescription = getString(R.string.cd_recents)
        root.setOnClickListener { if (recentsPanel == null) showRecents() else hideRecents() }

        val params = overlayParams(size, size, x, y)
        runCatching { wm.addView(root, params) }.onSuccess { chip = BubbleHandle(root, icon, params) }
    }

    private fun showRecents() {
        val recents = manager.recentsSnapshot()
        val anchor = chip ?: return
        if (recents.isEmpty()) return

        val inflater = LayoutInflater.from(themed)
        val panel = inflater.inflate(R.layout.view_recents_panel, null)
        val list = panel.findViewById<LinearLayout>(R.id.recents_list)
        recents.forEach { pkg ->
            val row = inflater.inflate(R.layout.item_app, list, false)
            row.findViewById<ImageView>(R.id.app_icon)
                .setImageDrawable(runCatching { packageManager.getApplicationIcon(pkg) }.getOrNull())
            row.findViewById<TextView>(R.id.app_label).text = labelOf(pkg)
            row.findViewById<TextView>(R.id.app_package).visibility = View.GONE
            row.setOnClickListener {
                hideRecents()
                openBubble(pkg)
            }
            list.addView(row)
        }
        panel.setOnTouchListener { _, event ->
            if (event.action == MotionEvent.ACTION_OUTSIDE) {
                hideRecents()
                true
            } else {
                false
            }
        }
        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_WATCH_OUTSIDE_TOUCH,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.END
            x = dp(BUBBLE_DP + 8)
            y = anchor.params.y
        }
        runCatching { wm.addView(panel, params) }.onSuccess { recentsPanel = panel }
    }

    private fun hideRecents() {
        recentsPanel?.let { runCatching { wm.removeView(it) } }
        recentsPanel = null
    }

import android.content.Intent

import kotlin.concurrent.thread

/** Bottom sheet listing launchable apps with a search field. */
class AppPickerSheet(
    private val activity: AppCompatActivity,
    private val onPick: (packageName: String) -> Unit
) {
    private class Entry(val packageName: String, val label: String, val icon: Drawable)

    fun show() {
        val dialog = BottomSheetDialog(activity)
        val binding = SheetAppPickerBinding.inflate(LayoutInflater.from(activity))
        val adapter = Adapter { entry ->
            dialog.dismiss()
            onPick(entry.packageName)
        }
        binding.list.layoutManager = LinearLayoutManager(activity)
        binding.list.adapter = adapter
        binding.search.doAfterTextChanged { adapter.filter(it?.toString().orEmpty()) }

        dialog.setContentView(binding.root)
        dialog.behavior.skipCollapsed = true
        dialog.behavior.state = BottomSheetBehavior.STATE_EXPANDED
        dialog.show()

        thread {
            val apps = loadApps()
            activity.runOnUiThread { adapter.submit(apps) }
        }
    }

    private fun loadApps(): List<Entry> {
        val pm = activity.packageManager
        val launcher = Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER)
        return pm.queryIntentActivities(launcher, 0)
            .filter { it.activityInfo.packageName != activity.packageName }
            .distinctBy { it.activityInfo.packageName }
            .map { Entry(it.activityInfo.packageName, it.loadLabel(pm).toString(), it.loadIcon(pm)) }
            .sortedBy { it.label.lowercase() }
    }

    private class Adapter(
        private val onClick: (Entry) -> Unit
    ) : RecyclerView.Adapter<Adapter.Holder>() {

        private var all: List<Entry> = emptyList()
        private var shown: List<Entry> = emptyList()
        private var query = ""

        class Holder(val binding: ItemAppBinding) : RecyclerView.ViewHolder(binding.root)

        fun submit(items: List<Entry>) {
            all = items
            filter(query)
        }

        fun filter(text: String) {
            query = text
            shown = if (text.isBlank()) all else all.filter { it.label.contains(text, ignoreCase = true) }
            notifyDataSetChanged()
        }

        override fun onCreateViewHolder(parent: ViewGroup, viewType: Int) =
            Holder(ItemAppBinding.inflate(LayoutInflater.from(parent.context), parent, false))

        override fun getItemCount() = shown.size

        override fun onBindViewHolder(holder: Holder, position: Int) {
            val entry = shown[position]
            holder.binding.appIcon.setImageDrawable(entry.icon)
            holder.binding.appLabel.text = entry.label
            holder.binding.appPackage.text = entry.packageName
            holder.binding.root.setOnClickListener { onClick(entry) }
        }
    }
}

package com.multitask.bubble.ui

import android.content.Context
import android.graphics.PixelFormat
import android.graphics.Point
import android.hardware.display.DisplayManager
import android.hardware.display.VirtualDisplay
import android.os.Handler
import android.os.Looper
import android.os.SystemClock
import android.view.Gravity
import android.view.KeyEvent
import android.view.LayoutInflater
import android.view.MotionEvent
import android.view.SurfaceHolder
import android.view.SurfaceView
import android.view.View
import android.view.ViewConfiguration
import android.view.WindowManager
import android.widget.ImageView
import android.widget.TextView
import com.multitask.bubble.R
import com.multitask.bubble.core.DiagLog
import com.multitask.bubble.core.WindowBackend
import java.util.concurrent.Executor
import java.util.concurrent.Executors

/**
 * The real bubble: the app runs on a private virtual display and its pixels are drawn into a
 * SurfaceView inside our own overlay window. We own the frame, so minimize and close are real.
 * Needs only the public DisplayManager API plus Shizuku for launching the app and injecting input.
 *
 * [onFailed] fires once if the device cannot run apps on a secondary display, so the caller can
 * switch to the freeform fallback.
 */
class EmbeddedWindow(
    private val context: Context,
    private val wm: WindowManager,
    private val backend: WindowBackend,
    private val worker: Executor,
    private val pkg: String,
    private val onMinimize: () -> Unit,
    private val onClose: () -> Unit,
    private val onFailed: () -> Unit
) : BubbleWindow, SurfaceHolder.Callback {

    private companion object {
        const val TAG = "EmbeddedWindow"
        const val FLAG_SUPPORTS_TOUCH = 1 shl 6 // hidden DisplayManager flag
        const val WIDTH_FRACTION = 0.8f
        const val HEIGHT_FRACTION = 0.6f
    }

    private val ui = Handler(Looper.getMainLooper())
    private val metrics = context.resources.displayMetrics
    private val surfaceWidth = (metrics.widthPixels * WIDTH_FRACTION).toInt()
    private val surfaceHeight = (metrics.heightPixels * HEIGHT_FRACTION).toInt()
    private val input = Executors.newSingleThreadExecutor()
    private val position = Point((metrics.widthPixels * 0.04f).toInt(), (metrics.heightPixels * 0.1f).toInt())

    private var root: View? = null
    private var virtualDisplay: VirtualDisplay? = null
    private var displayId = -1
    private var downTime = 0L
    private var released = false
    private var failed = false

    override fun show() {
        if (root != null || released) return
        val view = LayoutInflater.from(context).inflate(R.layout.view_embedded_window, null)

        val surface = view.findViewById<SurfaceView>(R.id.surface)
        surface.layoutParams = surface.layoutParams.apply {
            width = surfaceWidth
            height = surfaceHeight
        }
        surface.holder.addCallback(this)
        surface.setOnTouchListener { _, event ->
            forward(event)
            true
        }

        val pm = context.packageManager
        view.findViewById<TextView>(R.id.title_text).text = runCatching {
            pm.getApplicationLabel(pm.getApplicationInfo(pkg, 0)).toString()
        }.getOrDefault(pkg)
        view.findViewById<ImageView>(R.id.title_icon)
            .setImageDrawable(runCatching { pm.getApplicationIcon(pkg) }.getOrNull())
        view.findViewById<View>(R.id.btn_back).setOnClickListener { sendKey(KeyEvent.KEYCODE_BACK) }
        view.findViewById<View>(R.id.btn_minimize).setOnClickListener { onMinimize() }
        view.findViewById<View>(R.id.btn_close).setOnClickListener { onClose() }

        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.START
            x = position.x
            y = position.y
        }
        attachTitleDrag(view.findViewById(R.id.title_bar), view, params)

        try {
            wm.addView(view, params)
            root = view
        } catch (e: Exception) {
            DiagLog.log(TAG, "Could not show window for $pkg: ${e.message}")
        }
    }

    override fun hide() {
        root?.let { runCatching { wm.removeView(it) } }
        root = null
        virtualDisplay?.surface = null // pauses drawing; the app keeps running
    }

    override fun release() {
        released = true
        hide()
        val display = virtualDisplay
        virtualDisplay = null
        displayId = -1
        input.shutdown()
        worker.execute {
            // Stop the app first, otherwise Android moves it to the main screen when the display goes.
            backend.close(pkg)
            Thread.sleep(200)
            display?.release()
        }
    }

    // ---- surface lifecycle

    override fun surfaceCreated(holder: SurfaceHolder) {
        val existing = virtualDisplay
        if (existing != null) {
            existing.surface = holder.surface
            return
        }
        val manager = context.getSystemService(DisplayManager::class.java)
        val base = DisplayManager.VIRTUAL_DISPLAY_FLAG_PUBLIC or DisplayManager.VIRTUAL_DISPLAY_FLAG_OWN_CONTENT_ONLY
        val created = runCatching {
            manager.createVirtualDisplay(
                "bubble-$pkg", surfaceWidth, surfaceHeight, metrics.densityDpi,
                holder.surface, base or FLAG_SUPPORTS_TOUCH
            )
        }.recoverCatching {
            manager.createVirtualDisplay(
                "bubble-$pkg", surfaceWidth, surfaceHeight, metrics.densityDpi, holder.surface, base
            )
        }.onFailure { DiagLog.log(TAG, "Virtual display failed: ${it.message}") }.getOrNull()

        if (created == null) {
            fail()
            return
        }
        virtualDisplay = created
        displayId = created.display.displayId
        DiagLog.log(TAG, "$pkg runs on virtual display $displayId (${surfaceWidth}x$surfaceHeight)")
        val id = displayId
        worker.execute {
            if (!backend.launchOnDisplay(pkg, id)) ui.post { fail() }
        }
    }

    override fun surfaceChanged(holder: SurfaceHolder, format: Int, width: Int, height: Int) = Unit

    override fun surfaceDestroyed(holder: SurfaceHolder) {
        virtualDisplay?.surface = null
    }

    private fun fail() {
        if (failed || released) return
        failed = true
        DiagLog.log(TAG, "Embedding is not supported here, switching to freeform windows")
        onFailed()
    }

    // ---- input

    private fun forward(event: MotionEvent) {
        val id = displayId
        if (id <= 0) return
        val action = when (event.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                downTime = SystemClock.uptimeMillis()
                MotionEvent.ACTION_DOWN
            }
            MotionEvent.ACTION_MOVE, MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> event.actionMasked
            else -> return // single pointer for now
        }
        val x = event.x
        val y = event.y
        val time = downTime
        runCatching { input.execute { backend.injectMotion(id, action, x, y, time) } }
    }

    private fun sendKey(keyCode: Int) {
        val id = displayId
        if (id <= 0) return
        runCatching { input.execute { backend.injectKey(id, keyCode) } }
    }

    private fun attachTitleDrag(handle: View, window: View, params: WindowManager.LayoutParams) {
        val slop = ViewConfiguration.get(context).scaledTouchSlop
        var startX = 0
        var startY = 0
        var downX = 0f
        var downY = 0f
        handle.setOnTouchListener { _, event ->
            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    startX = params.x
                    startY = params.y
                    downX = event.rawX
                    downY = event.rawY
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val dx = (event.rawX - downX).toInt()
                    val dy = (event.rawY - downY).toInt()
                    if (Math.abs(dx) > slop || Math.abs(dy) > slop) {
                        params.x = startX + dx
                        params.y = startY + dy
                        position.set(params.x, params.y)
                        wm.updateViewLayout(window, params)
                    }
                    true
                }
                else -> true
            }
        }
    }
}

package com.multitask.bubble.ui

import android.content.Context
import android.graphics.Rect
import com.multitask.bubble.core.WindowBackend
import java.util.concurrent.Executor

/**
 * Fallback for devices that cannot run apps on a secondary display. The system draws the window,
 * we only place it. Hiding moves it off screen, which some ROMs may ignore.
 */
class FreeformWindow(
    context: Context,
    private val backend: WindowBackend,
    private val worker: Executor,
    private val pkg: String
) : BubbleWindow {

    private val bounds: Rect
    private val offscreen = Rect(-3000, -3000, -2000, -2000)

    @Volatile
    private var taskId = -1

    init {
        val metrics = context.resources.displayMetrics
        val w = metrics.widthPixels
        val h = metrics.heightPixels
        bounds = Rect((w * 0.08).toInt(), (h * 0.12).toInt(), (w * 0.92).toInt(), (h * 0.72).toInt())
    }

    override fun show() {
        // Launching again reuses the existing task and brings it to the front.
        worker.execute {
            val id = backend.launchFreeform(pkg, bounds)
            if (id > 0) taskId = id
        }
    }

    override fun hide() {
        worker.execute { if (taskId > 0) backend.moveTask(taskId, offscreen) }
    }

    override fun release() {
        worker.execute { backend.close(pkg) }
    }
}

===== FILE: app/src/test/java/com/multitask/bubble/BubbleManagerTest.kt =====
package com.multitask.bubble

import android.content.Context
import androidx.test.core.app.ApplicationProvider
import com.multitask.bubble.core.BubbleManager
import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Test
import org.junit.runner.RunWith
import org.robolectric.RobolectricTestRunner
import org.robolectric.annotation.Config

@RunWith(RobolectricTestRunner::class)
@Config(sdk = [34])
class BubbleManagerTest {

    private val context: Context get() = ApplicationProvider.getApplicationContext()

    @Test
    fun capEvictsTheOldestBubble() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")
        manager.open("app.two")
        manager.open("app.three")
        assertEquals(BubbleManager.MAX_BUBBLES, manager.snapshot().size)

        val change = manager.open("app.four")

        assertEquals(listOf("app.one"), change.evicted)
        val pkgs = manager.snapshot().map { it.pkg }
        assertFalse(pkgs.contains("app.one"))
        assertEquals("app.one", manager.recentsSnapshot().first())
    }

    @Test
    fun onlyOneBubbleIsOpenAtATime() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")

        val change = manager.open("app.two")

        assertEquals(listOf("app.one"), change.collapsed)
        val state = manager.snapshot().associateBy { it.pkg }
        assertFalse(state.getValue("app.one").expanded)
        assertTrue(state.getValue("app.two").expanded)
    }

    @Test
    fun openingAnExistingAppDoesNotDuplicateIt() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")
        manager.open("app.one")
        assertEquals(1, manager.snapshot().size)
    }

    @Test
    fun collapseKeepsTheBubbleButClosesIt() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")

        manager.collapse("app.one")

        assertEquals(1, manager.snapshot().size)
        assertFalse(manager.snapshot().first().expanded)
    }

    @Test
    fun dismissedBubblesMoveToRecentsAndReopen() {
        val manager = BubbleManager(context) {}
        manager.open("app.one")
        manager.dismiss("app.one")
        assertTrue(manager.snapshot().isEmpty())
        assertEquals(listOf("app.one"), manager.recentsSnapshot())

        manager.open("app.one")
        assertEquals(1, manager.snapshot().size)
        assertTrue(manager.recentsSnapshot().isEmpty())
    }

    @Test
    fun bubblesAreRestoredCollapsedByANewManager() {
        BubbleManager(context) {}.open("app.one")

        val restored = BubbleManager(context) {}.snapshot()

        assertEquals(listOf("app.one"), restored.map { it.pkg })
        assertFalse(restored.first().expanded)
    }

    @Test
    fun changesNotifyTheListener() {
        var calls = 0
        val manager = BubbleManager(context) { calls++ }
        manager.open("app.one")
        manager.collapse("app.one")
        manager.dismiss("app.one")
        assertEquals(3, calls)
    }
}
